import { useEffect, useMemo, useState } from 'preact/hooks';
import { Launcher } from './Launcher';
import { ConversationList } from './ConversationList';
import { ChatWindow } from './ChatWindow';
import { LoginScreen } from './LoginScreen';
import { NewConversation } from './NewConversation';
import { ConversationInfo } from './ConversationInfo';
import { ProfileScreen } from './ProfileScreen';
import { CallOverlay } from './CallOverlay';
import { CallHistory } from './CallHistory';
import { IconClose, IconMinimize, IconUsers, IconPhone } from './Icons';
import { store } from '../state/store';
import type { Conversation, Message } from '../types';
import type { ApiClient } from '../services/api-client';
import type { SocketClient } from '../services/socket-client';
import type { CallController } from '../services/call-controller';

type View = 'list' | 'chat' | 'new' | 'info' | 'profile' | 'calls';

export function App(props: {
  api: ApiClient;
  socket: SocketClient;
  call: CallController;
  tenantSlug: string;
  onAuthed: () => Promise<void>;
  onLogout: () => void;
  initialResetToken?: string;
}) {
  const [, setTick] = useState(0);
  const [view, setView] = useState<View>('list');
  const [replyTo, setReplyTo] = useState<Message | null>(null);
  const [editing, setEditing] = useState<Message | null>(null);

  // Call UI state (mirrors CallController)
  const [callState, setCallState] = useState(props.call.getState());

  // Re-render on any store change.
  useEffect(() => store.subscribe(() => setTick((t) => t + 1)), []);
  // Re-render on any call state change.
  useEffect(() => props.call.subscribe(setCallState), [props.call]);

  // On mount + on window focus: refresh conversations so newly added
  // groups appear even if the tab was backgrounded when the socket event
  // was broadcast.
  useEffect(() => {
    const onFocus = () => {
      if (store.get().authStatus === 'authenticated') {
        void props.socket.refreshConversations();
      }
    };
    // Also refresh immediately after the app mounts.
    if (store.get().authStatus === 'authenticated') {
      void props.socket.refreshConversations();
    }
    window.addEventListener('focus', onFocus);
    document.addEventListener('visibilitychange', onFocus);
    return () => {
      window.removeEventListener('focus', onFocus);
      document.removeEventListener('visibilitychange', onFocus);
    };
  }, [props.socket]);

  const s = store.get();
  const meId = s.me?.id ?? 0;

  const features = useMemo(
    () => ({
      reaction: true,
      reply: true,
      delete: true,
      edit: true,
      forward: true,
      pin: true,
    }),
    [],
  );

  const activeConv: Conversation | null = s.activeConversationId
    ? s.conversations.find((c) => c.id === s.activeConversationId) ?? null
    : null;

  const activeMessages = activeConv
    ? s.messagesByConversation[activeConv.id] ?? []
    : [];

  const typingIds = activeConv
    ? [...(s.typingByConversation[activeConv.id] ?? new Set<number>())]
    : [];

  useEffect(() => {
    if (callState.call) return;
    if (activeConv && view !== 'info' && view !== 'profile' && view !== 'calls') {
      setView('chat');
    } else if (
      !activeConv &&
      view !== 'new' &&
      view !== 'profile' &&
      view !== 'calls'
    ) {
      setView('list');
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [activeConv?.id, callState.call]);

  // ---------------------------------------------------------------
  // History
  // ---------------------------------------------------------------

  async function fetchLatestHistory(conversationId: number): Promise<void> {
    store.set((prev) => ({
      historyLoadingByConversation: {
        ...prev.historyLoadingByConversation,
        [conversationId]: true,
      },
    }));
    try {
      const page = await props.api.history(conversationId, 40, null);
      store.setMessages(conversationId, page.items);
      store.set((prev) => ({
        historyCursorByConversation: {
          ...prev.historyCursorByConversation,
          [conversationId]: page.nextCursor,
        },
        historyHasMoreByConversation: {
          ...prev.historyHasMoreByConversation,
          [conversationId]: page.hasMore,
        },
      }));
    } finally {
      store.set((prev) => ({
        historyLoadingByConversation: {
          ...prev.historyLoadingByConversation,
          [conversationId]: false,
        },
      }));
    }
  }

  async function handleOpenConversation(id: number) {
    store.set({ activeConversationId: id });
    store.clearUnread(id);
    props.socket.joinConversation(id);
    setView('chat');
    await fetchLatestHistory(id);

    const msgs = store.get().messagesByConversation[id] ?? [];
    const unreadIds = msgs.filter((m) => m.senderId !== meId).map((m) => m.id);
    if (unreadIds.length > 0) props.socket.markRead(id, unreadIds.slice(-50));
  }

  async function loadOlder() {
    if (!activeConv) return;
    const hasMore = s.historyHasMoreByConversation[activeConv.id] ?? false;
    if (!hasMore) return;
    const cursor = s.historyCursorByConversation[activeConv.id] ?? null;
    const page = await props.api.history(activeConv.id, 40, cursor);
    store.prependMessages(activeConv.id, page.items);
    store.set((prev) => ({
      historyCursorByConversation: {
        ...prev.historyCursorByConversation,
        [activeConv.id]: page.nextCursor,
      },
      historyHasMoreByConversation: {
        ...prev.historyHasMoreByConversation,
        [activeConv.id]: page.hasMore,
      },
    }));
  }

  // ---------------------------------------------------------------
  // Actions
  // ---------------------------------------------------------------

  function sendMessage(text: string, attachmentIds: number[], type?: Message['type']) {
    if (!activeConv) return;
    const clientId = `c_${Date.now()}_${Math.random().toString(36).slice(2)}`;
    props.socket.sendMessage(
      {
        conversationId: activeConv.id,
        content: text,
        type: type ?? (attachmentIds.length > 0 ? 'file' : 'text'),
        replyToId: replyTo?.id ?? null,
        clientId,
        attachmentIds,
      },
      (res) => {
        if (!res?.ok && res?.error) {
          store.set({ error: `Send failed: ${res.error}` });
        }
        setReplyTo(null);
      },
    );
  }

  function deleteMessage(m: Message) {
    if (!confirm('Delete this message?')) return;
    props.socket.deleteMessage(m.id);
  }

  function submitEdit(text: string) {
    if (!editing) return;
    props.socket.editMessage(editing.id, text);
    setEditing(null);
  }

  function reactToMessage(m: Message, reaction: string, added: boolean) {
    props.socket.react(m.id, reaction, added);
  }

  async function handleForward(m: Message) {
    const list = s.conversations.filter((c) => c.id !== m.conversationId);
    if (list.length === 0) {
      store.set({ error: 'No other conversations to forward to' });
      return;
    }
    const choice = prompt(
      `Forward to:\n${list
        .map((c) => `${c.id} = ${c.name ?? 'Direct #' + c.id}`)
        .join('\n')}\n\nEnter conversation id:`,
    );
    const target = Number(choice);
    if (!Number.isFinite(target) || target <= 0) return;
    try {
      await props.api.forwardMessage(m.id, [target]);
    } catch (err) {
      store.set({ error: err instanceof Error ? err.message : String(err) });
    }
  }

  async function handlePin(m: Message, pinned: boolean) {
    try {
      await props.api.pinMessage(m.id, pinned);
    } catch (err) {
      store.set({ error: err instanceof Error ? err.message : String(err) });
    }
  }

  async function startAudioCall() {
    if (!activeConv) return;
    if (callState.call) {
      store.set({ error: 'Already in a call' });
      return;
    }
    try {
      await props.call.startOutgoing(activeConv.id, 'audio');
    } catch (err) {
      store.set({ error: err instanceof Error ? err.message : String(err) });
    }
  }

  async function startVideoCall() {
    if (!activeConv) return;
    if (callState.call) {
      store.set({ error: 'Already in a call' });
      return;
    }
    try {
      await props.call.startOutgoing(activeConv.id, 'video');
    } catch (err) {
      store.set({ error: err instanceof Error ? err.message : String(err) });
    }
  }

  // ---------------------------------------------------------------
  // Derived
  // ---------------------------------------------------------------

  const isAuthed = s.authStatus === 'authenticated';
  const needsAuth = s.authStatus === 'needs-auth';
  const totalUnread = Object.values(s.unreadByConversation).reduce(
    (a, b) => a + b,
    0,
  );

  const callPeerLabel = (() => {
    const c = callState.call;
    if (!c) return 'Call';
    const other = c.participants.find((p) => p.userId !== meId);
    const conv = s.conversations.find((cc) => cc.id === c.conversationId);
    const otherMember = conv?.members.find((m) => m.userId === other?.userId);
    return otherMember?.displayName ?? `User #${other?.userId ?? '?'}`;
  })();

  const firstRemoteStream = callState.remoteStreams.size
    ? [...callState.remoteStreams.values()][0]!
    : null;

  // ---------------------------------------------------------------
  // Render
  // ---------------------------------------------------------------

  return (
    <>
      {!s.isOpen && !callState.call && (
        <Launcher
          open={false}
          unread={totalUnread}
          onClick={() => store.set({ isOpen: true })}
        />
      )}

      {s.isOpen && !callState.call && (
        <div class="cw-panel" role="dialog" aria-label="Chat widget">
          <header class="cw-header">
            <div>
              <div class="cw-header__title">Chat</div>
              <div class="cw-header__sub">
                {isAuthed && s.me
                  ? s.me.displayName
                  : needsAuth
                  ? 'Sign in required'
                  : 'Connecting…'}
              </div>
            </div>
            <div class="cw-header__actions">
              {isAuthed && (
                <>
                  <button
                    type="button"
                    class="cw-icon-btn"
                    aria-label="Call history"
                    title="Call history"
                    onClick={() => setView('calls')}
                  >
                    <IconPhone />
                  </button>
                  <button
                    type="button"
                    class="cw-icon-btn"
                    aria-label="Profile"
                    title="Profile"
                    onClick={() => setView('profile')}
                  >
                    <IconUsers />
                  </button>
                </>
              )}
              <button
                type="button"
                class="cw-icon-btn"
                aria-label="Minimize"
                onClick={() => store.set({ isOpen: false })}
              >
                <IconMinimize />
              </button>
              <button
                type="button"
                class="cw-icon-btn"
                aria-label="Close"
                onClick={() => store.set({ isOpen: false })}
              >
                <IconClose />
              </button>
            </div>
          </header>

          {s.error && (
            <div
              class="cw-errorbar"
              onClick={() => store.set({ error: null })}
            >
              {s.error}
            </div>
          )}

          {needsAuth && (
            <LoginScreen
              api={props.api}
              socket={props.socket}
              tenantSlug={props.tenantSlug}
              onAuthed={props.onAuthed}
              initialResetToken={props.initialResetToken}
            />
          )}

          {isAuthed && view === 'list' && (
            <ConversationList
              conversations={s.conversations}
              meId={meId}
              activeId={s.activeConversationId}
              onSelect={handleOpenConversation}
              onNewChat={() => setView('new')}
            />
          )}

          {isAuthed && view === 'new' && (
            <NewConversation
              api={props.api}
              onBack={() => setView('list')}
              onCreated={async (id) => {
                // Refresh list, then open the created conversation.
                await props.socket.refreshConversations();
                await handleOpenConversation(id);
              }}
            />
          )}

          {isAuthed && view === 'chat' && activeConv && (
            <ChatWindow
              conversation={activeConv}
              meId={meId}
              messages={activeMessages}
              typingUserIds={typingIds}
              replyTo={replyTo}
              editing={editing}
              features={features}
              api={props.api}
              onBack={() => {
                store.set({ activeConversationId: null });
                setReplyTo(null);
                setEditing(null);
                setView('list');
              }}
              onShowInfo={() => setView('info')}
              onSend={sendMessage}
              onTypingStart={() => props.socket.startTyping(activeConv.id)}
              onTypingStop={() => props.socket.stopTyping(activeConv.id)}
              onReply={setReplyTo}
              onCancelReply={() => setReplyTo(null)}
              onEdit={setEditing}
              onCancelEdit={() => setEditing(null)}
              onSubmitEdit={submitEdit}
              onDelete={deleteMessage}
              onReact={reactToMessage}
              onForward={handleForward}
              onPin={handlePin}
              onLoadOlder={loadOlder}
              onStartAudioCall={startAudioCall}
              onStartVideoCall={startVideoCall}
              onError={(msg) => store.set({ error: msg })}
            />
          )}

          {isAuthed && view === 'info' && activeConv && (
            <ConversationInfo
              conversation={activeConv}
              api={props.api}
              meId={meId}
              onBack={() => setView('chat')}
              onLeft={() => {
                store.set({ activeConversationId: null });
                setView('list');
              }}
              onUpdated={(c) => {
                store.upsertConversation(c);
                // Also refresh the full list to pick up any member changes.
                void props.socket.refreshConversations();
              }}
            />
          )}

          {isAuthed && view === 'profile' && (
            <ProfileScreen
              api={props.api}
              onBack={() => setView(activeConv ? 'chat' : 'list')}
              onLogout={props.onLogout}
            />
          )}

          {isAuthed && view === 'calls' && (
            <CallHistory api={props.api} onBack={() => setView('list')} />
          )}

          {s.isLoading && !needsAuth && (
            <div class="cw-empty">
              <div class="cw-spinner" />
            </div>
          )}
        </div>
      )}

      {callState.call && callState.status !== 'idle' && (
        <CallOverlay
          call={callState.call}
          direction={callState.direction ?? 'outgoing'}
          status={callState.status}
          localStream={callState.localStream}
          remoteStream={firstRemoteStream}
          micMuted={callState.micMuted}
          cameraOn={callState.cameraOn}
          quality={callState.quality}
          peerLabel={callPeerLabel}
          onAccept={() => void props.call.accept()}
          onReject={() => void props.call.reject()}
          onHangUp={() => void props.call.hangUp()}
          onToggleMic={() => props.call.toggleMic()}
          onToggleCamera={() => void props.call.toggleCamera()}
          onSwitchCamera={() => void props.call.switchCamera()}
        />
      )}
    </>
  );
}

import { useState } from 'preact/hooks';
import type { Message, ConversationMember } from '../types';
import { MessageMenu } from './MessageMenu';
import { Lightbox, type LightboxMedia } from './Lightbox';
import { formatLocalTime, humanSize } from '../utils/time';

interface Attachment {
  id: number;
  mimeType: string;
  url: string;
  sizeBytes: number;
  originalName: string | null;
}

export function MessageBubble(props: {
  msg: Message;
  isMine: boolean;
  meId: number;
  sender: ConversationMember | undefined;
  isGroup: boolean;
  canModerate: boolean;
  features: {
    reply: boolean;
    edit: boolean;
    delete: boolean;
    reaction: boolean;
    forward: boolean;
    pin: boolean;
  };
  onReply: (m: Message) => void;
  onDelete: (m: Message) => void;
  onEdit: (m: Message) => void;
  onReact: (m: Message, reaction: string, added: boolean) => void;
  onForward: (m: Message) => void;
  onPin: (m: Message, pinned: boolean) => void;
}) {
  const [menuOpen, setMenuOpen] = useState(false);
  const [lightbox, setLightbox] = useState<LightboxMedia | null>(null);

  const { msg, isMine } = props;
  const mineReactions = new Set(
    (msg.reactions ?? [])
      .filter((r) => r.userId === props.meId)
      .map((r) => r.reaction),
  );
  const counts = new Map<string, number>();
  for (const r of msg.reactions ?? []) {
    counts.set(r.reaction, (counts.get(r.reaction) ?? 0) + 1);
  }

  const isDeleted = Boolean(msg.deletedAt);
  const attachments = (msg.attachments ?? []) as unknown as Attachment[];

  const senderName = props.sender?.displayName ?? `User #${msg.senderId ?? '?'}`;
  const initials = senderName
    .split(/\s+/)
    .map((s) => s[0] || '')
    .join('')
    .slice(0, 2)
    .toUpperCase();

  return (
    <div class={`cw-msgRow ${isMine ? 'cw-msgRow--me' : 'cw-msgRow--them'}`}>
      {!isMine && props.isGroup && (
        <div class="cw-msgRow__avatar" title={senderName}>
          {props.sender?.avatarUrl ? (
            <img src={props.sender.avatarUrl} alt={senderName} />
          ) : (
            <span>{initials}</span>
          )}
        </div>
      )}

      <div
        class={`cw-msg ${isMine ? 'cw-msg--me' : 'cw-msg--them'} ${
          isDeleted ? 'cw-msg--deleted' : ''
        }`}
      >
        {!isMine && props.isGroup && (
          <div class="cw-msg__sender">{senderName}</div>
        )}

        {msg.replyToId && (
          <div class="cw-msg__reply">Reply to #{msg.replyToId}</div>
        )}

        {msg.pinnedAt && <div class="cw-msg__pinned">📌 Pinned</div>}
        {msg.forwardedFromId && (
          <div class="cw-msg__forwarded">↗ Forwarded</div>
        )}

        {!isDeleted &&
          attachments.map((a) => (
            <AttachmentView
              key={a.id}
              attachment={a}
              onOpen={(media) => setLightbox(media)}
            />
          ))}

        {!isDeleted && msg.content && (
          <div class="cw-msg__text">{msg.content}</div>
        )}
        {isDeleted && (
          <div class="cw-msg__text">This message was deleted</div>
        )}

        <div class="cw-msg__meta">{formatLocalTime(msg.createdAt)}
          {msg.editedAt ? ' · edited' : ''}
        </div>

        {!isDeleted && counts.size > 0 && (
          <div class="cw-msg__reactions">
            {[...counts.entries()].map(([reaction, count]) => (
              <button
                key={reaction}
                type="button"
                class="cw-reaction"
                data-mine={mineReactions.has(reaction)}
                onClick={() =>
                  props.onReact(msg, reaction, !mineReactions.has(reaction))
                }
              >
                <span>{reaction}</span>
                <span>{count}</span>
              </button>
            ))}
          </div>
        )}

        {!isDeleted && (
          <button
            type="button"
            class="cw-msg__more"
            onClick={(e) => {
              e.stopPropagation();
              setMenuOpen((v) => !v);
            }}
            aria-label="Message options"
            title="Message options"
          >
            ⋮
          </button>
        )}

        {menuOpen && (
          <MessageMenu
            msg={msg}
            isMine={isMine}
            isGroup={props.isGroup}
            canModerate={props.canModerate}
            features={props.features}
            onClose={() => setMenuOpen(false)}
            onReply={() => props.onReply(msg)}
            onEdit={() => props.onEdit(msg)}
            onDelete={() => props.onDelete(msg)}
            onReact={(reaction) =>
              props.onReact(msg, reaction, !mineReactions.has(reaction))
            }
            onCopy={() => {
              const text = msg.content ?? '';
              if (text) void navigator.clipboard.writeText(text);
            }}
            onForward={() => props.onForward(msg)}
            onPin={() => props.onPin(msg, !msg.pinnedAt)}
          />
        )}
      </div>

      <Lightbox media={lightbox} onClose={() => setLightbox(null)} />
    </div>
  );
}

function AttachmentView({
  attachment,
  onOpen,
}: {
  attachment: Attachment;
  onOpen: (m: LightboxMedia) => void;
}) {
  const mime = attachment.mimeType;

  if (mime.startsWith('image/')) {
    return (
      <div class="cw-attach cw-attach--image">
        <button
          type="button"
          class="cw-attach__open"
          onClick={() =>
            onOpen({
              kind: 'image',
              url: attachment.url,
              name: attachment.originalName,
            })
          }
        >
          <img
            src={attachment.url}
            alt={attachment.originalName ?? 'image'}
            loading="lazy"
          />
        </button>
      </div>
    );
  }

  if (mime.startsWith('video/')) {
    return (
      <div class="cw-attach cw-attach--video">
        <button
          type="button"
          class="cw-attach__open"
          onClick={() =>
            onOpen({
              kind: 'video',
              url: attachment.url,
              name: attachment.originalName,
            })
          }
        >
          <video src={attachment.url} preload="metadata" />
          <span class="cw-attach__play">▶</span>
        </button>
      </div>
    );
  }

  if (mime.startsWith('audio/')) {
    return (
      <div class="cw-attach cw-attach--audio">
        <audio src={attachment.url} controls preload="metadata" />
      </div>
    );
  }

  return (
    <a
      class="cw-attach cw-attach--file"
      href={attachment.url}
      target="_blank"
      rel="noreferrer"
      download
    >
      <span>📄</span>
      <span class="cw-attach__name">
        {attachment.originalName ?? 'file'}
      </span>
      <span class="cw-attach__size">
        {humanSize(attachment.sizeBytes)}
      </span>
    </a>
  );
}

import { useEffect, useRef, useState } from 'preact/hooks';
import type { Message } from '../types';
import { IconEdit, IconReply, IconTrash, IconCheck } from './Icons';

const REACTIONS = ['👍', '❤️', '😂', '😮', '😢', '🙏'];

export function MessageMenu(props: {
  msg: Message;
  isMine: boolean;
  isGroup: boolean;
  canModerate: boolean;
  features: {
    reply: boolean;
    edit: boolean;
    delete: boolean;
    reaction: boolean;
    forward: boolean;
    pin: boolean;
  };
  onClose: () => void;
  onReply: () => void;
  onEdit: () => void;
  onDelete: () => void;
  onReact: (reaction: string) => void;
  onCopy: () => void;
  onForward: () => void;
  onPin: () => void;
}) {
  const [showReactions, setShowReactions] = useState(false);
  const ref = useRef<HTMLDivElement | null>(null);

  // Close on outside click / Escape. Capture the callback in a ref so the
  // effect doesn't re-bind on every render.
  const closeRef = useRef(props.onClose);
  useEffect(() => {
    closeRef.current = props.onClose;
  }, [props.onClose]);

  useEffect(() => {
    const onClick = (e: MouseEvent) => {
      if (ref.current && !ref.current.contains(e.target as Node)) {
        closeRef.current();
      }
    };
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'Escape') closeRef.current();
    };
    document.addEventListener('mousedown', onClick);
    document.addEventListener('keydown', onKey);
    return () => {
      document.removeEventListener('mousedown', onClick);
      document.removeEventListener('keydown', onKey);
    };
  }, []);

  const isDeleted = Boolean(props.msg.deletedAt);

  // Helper to keep click handlers tidy
  const invoke = (fn: () => void) => (e: Event) => {
    e.stopPropagation();
    fn();
    props.onClose();
  };

  return (
    <div
      ref={ref}
      class="cw-menu"
      role="menu"
      onClick={(e) => e.stopPropagation()}
    >
      {props.features.reaction && !isDeleted && (
        <div class="cw-menu__reactions">
          {!showReactions ? (
            <button
              type="button"
              class="cw-menu__item cw-menu__item--react"
              onClick={(e) => {
                e.stopPropagation();
                setShowReactions(true);
              }}
            >
              🙂 React…
            </button>
          ) : (

        rows={1}
        style={{ flex: 1 }}
      />

        )}

        {filtered.map((c) => {

          const initials = initialsOf(title);
          const preview = previewText(c);

          return (

            >

                {avatarUrl ? <img src={avatarUrl} alt={title} /> : initials}
              </div>

function getTitle(c: Conversation, meId: number): string {
  if (c.type === 'group') return c.name ?? 'Group';
  const other = c.members.find((m) => m.userId !== meId);
  return other?.displayName ?? `User #${other?.userId ?? '?'}`;
}

function getAvatar(c: Conversation, meId: number): string | null {
  if (c.type === 'group') return c.avatarUrl ?? null;
  const other = c.members.find((m) => m.userId !== meId);
  return other?.avatarUrl ?? null;
}

function initialsOf(title: string): string {
  return title.split(/\s+/).map((s) => s[0] || '').join('').slice(0, 2).toUpperCase();

    case 'audio': return '🎵 Audio';

  }

        const items = await props.api.listUsers(term);
        const existing = new Set(props.conversation.members.map((m) => m.userId));

function getOrCreateDeviceId(): string {
  const KEY = 'chat_widget_device_id';
  try {
    let id = localStorage.getItem(KEY);
    if (!id) {
      id = `w_${Math.random().toString(36).slice(2)}_${Date.now()}`;
      localStorage.setItem(KEY, id);
    }
    return id;
  } catch { return `w_${Math.random().toString(36).slice(2)}`; }
}

import { io, type Socket } from 'socket.io-client';
import { store } from '../state/store';
import { bus } from './event-bus';
import { notifications } from './notification-service';
import type { Conversation, Message } from '../types';

export class SocketClient {
  private _socket: Socket | null = null;
  private baseUrl: string;

  constructor(baseUrl: string) {
    this.baseUrl = baseUrl.replace(/\/api\/v1\/?$/, '');
  }

  get socket(): Socket | null {
    return this._socket;
  }

  // ---------------------------------------------------------------
  // Connect / disconnect
  // ---------------------------------------------------------------

  connect(): void {
    const token = store.get().accessToken;
    if (!token) throw new Error('No access token');

    // Tear down any previous socket to avoid duplicate listeners.
    if (this._socket) {
      this._socket.removeAllListeners();
      this._socket.disconnect();
      this._socket = null;
    }

    this._socket = io(this.baseUrl, {
      path: '/socket.io',
      transports: ['websocket', 'polling'],
      withCredentials: true,
      auth: { token, deviceId: this.getDeviceId() },
      reconnection: true,
      reconnectionDelay: 1000,
      reconnectionDelayMax: 10_000,
      reconnectionAttempts: Infinity,
    });

    this._socket.on('connect', () => {
      // Join rooms for every conversation we already know about.
      const convs = store.get().conversations;
      for (const c of convs) {
        this._socket!.emit('conversation:join', { conversationId: c.id });
      }
      // Refresh conversations too — catches groups added while we were
      // offline or during the initial connect race.
      void this.refreshConversations();
    });

    this._socket.on('connect_error', async (err) => {
      if (String(err?.message || '').toUpperCase().includes('UNAUTHORIZED')) {
        const { ApiClient } = await import('./api-client');
        const baseApi = `${this.baseUrl.replace(/\/$/, '')}/api/v1`;
        const api = new ApiClient(baseApi);
        const ok = await api.refreshTokens();
        if (ok) this._socket!.auth = { token: store.get().accessToken };
      }
    });

    this.bindEvents();
  }

  disconnect(): void {
    if (this._socket) {
      this._socket.removeAllListeners();
      this._socket.disconnect();
      this._socket = null;
    }
  }

  // ---------------------------------------------------------------
  // Public helpers
  // ---------------------------------------------------------------

  joinConversation(conversationId: number): void {
    this._socket?.emit('conversation:join', { conversationId });
  }

  sendMessage(
    input: {
      conversationId: number;
      content: string;
      type?: Message['type'];
      replyToId?: number | null;
      clientId?: string;
      attachmentIds?: number[];
    },
    ack?: (res: { ok?: boolean; message?: Message; error?: string }) => void,
  ): void {
    this._socket?.emit('message:send', input, ack);
  }

  editMessage(messageId: number, content: string): void {
    this._socket?.emit('message:edit', { messageId, content });
  }

  deleteMessage(messageId: number): void {
    this._socket?.emit('message:delete', { messageId });
  }

  startTyping(conversationId: number): void {
    this._socket?.emit('typing:start', { conversationId });
  }

  stopTyping(conversationId: number): void {
    this._socket?.emit('typing:stop', { conversationId });
  }

  markRead(conversationId: number, messageIds: number[]): void {
    this._socket?.emit('message:read', { conversationId, messageIds });
  }

  react(messageId: number, reaction: string, added: boolean): void {
    this._socket?.emit('message:react', { messageId, reaction, added });
  }

  /**
   * Refetch the conversation list from the API. Used on socket (re)connect,
   * when a new conversation is announced, and on window focus.
   */
  async refreshConversations(): Promise<void> {
    try {
      const { ApiClient } = await import('./api-client');
      const baseApi = `${this.baseUrl.replace(/\/$/, '')}/api/v1`;
      const api = new ApiClient(baseApi);
      const list = await api.listConversations();
      store.set({ conversations: list });
      // Ensure the socket is in the room for every conversation.
      for (const c of list) {
        this._socket?.emit('conversation:join', { conversationId: c.id });
      }
    } catch {
      /* silent — will retry on next event */
    }
  }

  // ---------------------------------------------------------------
  // Call helpers (thin wrappers)
  // ---------------------------------------------------------------

  acceptCall(callId: number): void {
    this._socket?.emit('call:accept', { callId });
  }

  rejectCall(callId: number, reason: 'rejected' | 'busy' = 'rejected'): void {
    this._socket?.emit('call:reject', { callId, reason });
  }

  cancelCall(callId: number): void {
    this._socket?.emit('call:cancel', { callId });
  }

  endCall(
    callId: number,
    reason: 'completed' | 'remote_ended' | 'network_error' = 'completed',
  ): void {
    this._socket?.emit('call:end', { callId, reason });
  }

  // ---------------------------------------------------------------
  // Private
  // ---------------------------------------------------------------

  private bindEvents(): void {
    if (!this._socket) return;

    // --- Messages --------------------------------------------------
    this._socket.on('message:new', (msg: Message) => {
      store.addMessage(msg);

      const me = store.get().me;
      const conv = store.get().conversations.find((c) => c.id === msg.conversationId);

      // If we don't know the conversation yet (e.g. just added to a group),
      // refetch everything so the chat list and members update.
      if (!conv) {
        void this.refreshConversations();
      } else {
        store.upsertConversation({
          ...conv,
          lastMessageAt: msg.createdAt,
          lastMessage: {
            id: msg.id,
            senderId: msg.senderId,
            type: msg.type,
            content: msg.content,
            deleted: false,
            createdAt: msg.createdAt,
          },
        });
      }

      if (me && msg.senderId !== me.id) {
        bus.emit('messageReceived', msg);
        notifications.notifyMessage(conv, msg);
        if (store.get().activeConversationId === msg.conversationId) {
          this.markRead(msg.conversationId, [msg.id]);
          store.clearUnread(msg.conversationId);
        }
      } else {
        bus.emit('messageSent', msg);
      }
      bus.emit('message', msg);
    });

    this._socket.on('message:updated', (msg: Message) => {
      store.updateMessage(msg.conversationId, msg);
    });

    this._socket.on(
      'message:deleted',
      (payload: { messageId: number; conversationId: number }) => {
        const existing = store.get().messagesByConversation[payload.conversationId] ?? [];
        const target = existing.find((m) => m.id === payload.messageId);
        if (target) {
          store.updateMessage(payload.conversationId, {
            ...target,
            content: null,
            deletedAt: new Date().toISOString(),
          });
        }
      },
    );

    this._socket.on(
      'message:reaction',
      (payload: {
        messageId: number;
        userId: number;
        reaction: string;
        added: boolean;
      }) => {
        const all = store.get().messagesByConversation;
        for (const convId of Object.keys(all)) {
          const list = all[Number(convId)] ?? [];
          const msg = list.find((m) => m.id === payload.messageId);
          if (!msg) continue;
          const reactions = [...(msg.reactions ?? [])];
          const idx = reactions.findIndex(
            (r) => r.userId === payload.userId && r.reaction === payload.reaction,
          );
          if (payload.added && idx === -1) {
            reactions.push({
              userId: payload.userId,
              reaction: payload.reaction,
              createdAt: new Date().toISOString(),
            });
          } else if (!payload.added && idx !== -1) {
            reactions.splice(idx, 1);
          }
          store.updateMessage(msg.conversationId, { ...msg, reactions });
          break;
        }
      },
    );

    this._socket.on(
      'typing:update',
      (payload: { conversationId: number; userId: number; isTyping: boolean }) => {
        store.setTyping(payload.conversationId, payload.userId, payload.isTyping);
      },
    );

    this._socket.on(
      'presence:update',
      (payload: { userId: number; status: string; lastSeenAt: string }) => {
        const online = payload.status === 'online' || payload.status === 'away';
        store.setOnline(payload.userId, online);
        bus.emit(online ? 'userOnline' : 'userOffline', payload);
      },
    );

    // --- Conversations ---------------------------------------------
    // Someone added me to a group, or started a direct chat with me.
    // Always refetch the full list so we have member names, unread counts,
    // and last message preview.
    this._socket.on('conversation:created', async (conv: Conversation) => {
      try {
        await this.refreshConversations();
      } catch {
        // Fallback: at least merge what the server sent us.
        store.upsertConversation(conv);
      }
      this._socket!.emit('conversation:join', { conversationId: conv.id });
      bus.emit('ready', { conversation: conv });
    });

    // Removed from a group (or I left).
    this._socket.on(
      'conversation:removed',
      (payload: { conversationId: number }) => {
        store.set((s) => ({
          conversations: s.conversations.filter((c) => c.id !== payload.conversationId),
          activeConversationId:
            s.activeConversationId === payload.conversationId
              ? null
              : s.activeConversationId,
        }));
      },
    );

    // --- Calls -----------------------------------------------------
    this._socket.on('call:mark-connected', (payload: { callId: number }) => {
      // The server acknowledges connection; no client action required.
      void payload;
    });

    // --- Errors ----------------------------------------------------
    this._socket.on('error', (err: { code: string; message: string }) => {
      bus.emit('error', err);
    });
  }

  private getDeviceId(): string {
    const KEY = 'chat_widget_device_id';
    try {
      let id = localStorage.getItem(KEY);
      if (!id) {
        id = `w_${Math.random().toString(36).slice(2)}_${Date.now()}`;
        localStorage.setItem(KEY, id);
      }
      return id;
    } catch {
      return `w_${Math.random().toString(36).slice(2)}`;
    }
  }
}

    const call = await this.api.initiateCall({ conversationId, callType });

    this.patch({ call });

    window.setTimeout(() => {

      if (this.state.call?.id === call.id && this.state.status === 'ringing') {

        void this.cancel();

      }

    }, 45_000);

  }

    if (!call || this.state.direction !== 'incoming') return;



    await this.refreshIceServers();

    const localStream = await this.getUserMedia(call.callType);

    this.patch({ localStream });



    this.socket.socket?.emit('call:accept', { callId: call.id });

    // We wait for it in \`handleOffer\`.

  }

  async switchCamera(): Promise<void> {

    // Mobile: cycle facingMode.

    const s = this.state.localStream;

    if (!s) return;

    const cur = s.getVideoTracks()[0];

    if (!cur) return;

    const currentFacing = cur.getSettings().facingMode ?? 'user';

    const nextFacing = currentFacing === 'user' ? 'environment' : 'user';

    const fresh = await navigator.mediaDevices.getUserMedia({

      audio: false,

      video: { facingMode: nextFacing },

    for (const [, pc] of this.pcs) {

      if (sender) await sender.replaceTrack(nextTrack);

    }

    // Update local stream

    s.addTrack(nextTrack);

    this.patch({ localStream: s });

  }

  private bindSocket(): void {

    const socket = this.socket.socket;

    if (!socket) return;



    socket.on('call:incoming', (payload: { call: CallDTO; from: number }) => {

      // Only ring if we're not already in a call.

      if (this.state.call) {

        socket.emit('call:reject', { callId: payload.call.id, reason: 'busy' });

        return;

      }

      this.patch({

        call: payload.call,

        direction: 'incoming',

        status: 'ringing',

        remoteStreams: new Map(),

      });

      bus.emit('callIncoming', payload);

    });



    socket.on('call:ringing', (dto: CallDTO) => {

      // Our own outgoing call confirmed

      this.patch({ call: dto, direction: 'outgoing', status: 'ringing' });

    });



    socket.on('call:accepted', (payload: { callId: number; userId: number; call: CallDTO }) => {

      const call = this.state.call;

      if (!call || call.id !== payload.callId) return;

      this.patch({ call: payload.call, status: 'connecting' });



      // As the caller, initiate the offer to the accepted participant.

      void this.sendOfferTo(payload.userId);

    });



    socket.on('call:rejected', (payload: { callId: number; userId: number; reason: string }) => {

      const call = this.state.call;

      if (!call || call.id !== payload.callId) return;

      // In 1:1, the call is over.

      if (call.participants.filter((p) => p.role === 'callee').length <= 1) {

        this.patch({ status: 'ended' });

        this.cleanupLocal();

      } else {

        // In group, just remove this participant from our peer list.

        this.closePeer(payload.userId);

      }

    });



    socket.on('call:cancelled', (payload: { callId: number; byUserId: number }) => {

      const call = this.state.call;

      if (!call || call.id !== payload.callId) return;

      this.cleanupLocal();

    });



    socket.on('call:ended', (payload: { callId: number; byUserId: number; reason: string }) => {

      const call = this.state.call;

      if (!call || call.id !== payload.callId) return;

      this.cleanupLocal();

    });



    socket.on('call:offer', async (payload: { callId: number; fromUserId: number; sdp: string }) => {

      await this.handleOffer(payload.callId, payload.fromUserId, payload.sdp);

    });

      callId: number;

      fromUserId: number;

      candidate: RTCIceCandidateInit;

    }) => {

      await this.handleIce(payload.fromUserId, payload.candidate);

  private async sendOfferTo(peerUserId: number): Promise<void> {

    const call = this.state.call;

    if (!call) return;



    const pc = await this.getOrCreatePeer(peerUserId);

    const offer = await pc.createOffer({

      offerToReceiveAudio: true,

      offerToReceiveVideo: call.callType === 'video',

    });

    await pc.setLocalDescription(offer);



    this.socket.socket?.emit('call:offer', {

      callId: call.id,

      toUserId: peerUserId,

      sdp: offer.sdp ?? '',

    });

  }



  private async handleOffer(callId: number, fromUserId: number, sdp: string): Promise<void> {

    const call = this.state.call;

    if (!call || call.id !== callId) return;



    // If we're the callee, we set state to 'connecting' on first offer.

    if (this.state.status === 'ringing') {

      this.patch({ status: 'connecting' });

    }



    const pc = await this.getOrCreatePeer(fromUserId);

    await pc.setRemoteDescription({ type: 'offer', sdp });



    // Flush any pending ICE candidates

    const queued = this.pendingCandidates.get(fromUserId) ?? [];

    for (const c of queued) {

      try { await pc.addIceCandidate(c); } catch { /* ignore */ }

    }

    this.pendingCandidates.delete(fromUserId);



    const answer = await pc.createAnswer();

    await pc.setLocalDescription(answer);



    this.socket.socket?.emit('call:answer', {

      callId,

      toUserId: fromUserId,

      sdp: answer.sdp ?? '',

    });

  }



  private async handleAnswer(fromUserId: number, sdp: string): Promise<void> {

    const pc = this.pcs.get(fromUserId);

    if (!pc) return;

    if (pc.signalingState === 'stable') return; // already applied

    await pc.setRemoteDescription({ type: 'answer', sdp });



    // Flush pending ICE

    const queued = this.pendingCandidates.get(fromUserId) ?? [];

    for (const c of queued) {

      try { await pc.addIceCandidate(c); } catch { /* ignore */ }

    }

    this.pendingCandidates.delete(fromUserId);

  }



  private async handleIce(fromUserId: number, candidate: RTCIceCandidateInit): Promise<void> {

    const pc = this.pcs.get(fromUserId);

    if (!pc || !pc.remoteDescription) {

      const q = this.pendingCandidates.get(fromUserId) ?? [];

      q.push(candidate);

      this.pendingCandidates.set(fromUserId, q);

      return;

    }

    try {

      await pc.addIceCandidate(candidate);

    } catch {

      /* ignore */

    }

  }



  private async getOrCreatePeer(peerUserId: number): Promise<RTCPeerConnection> {

    const existing = this.pcs.get(peerUserId);

    if (existing) return existing;



    const pc = new RTCPeerConnection({

      iceServers: this.iceServers,

      iceCandidatePoolSize: 4,

    });



    // Add local tracks

    const ls = this.state.localStream;

    if (ls) {

      for (const track of ls.getTracks()) {

        pc.addTrack(track, ls);

      }

    }



    // ICE candidate handler

    pc.onicecandidate = (e) => {

      if (!e.candidate) return;

      const call = this.state.call;

      if (!call) return;

      this.socket.socket?.emit('call:ice', {

        callId: call.id,

        toUserId: peerUserId,

        candidate: {

          candidate: e.candidate.candidate,

          sdpMid: e.candidate.sdpMid,

          sdpMLineIndex: e.candidate.sdpMLineIndex,

          usernameFragment: e.candidate.usernameFragment,

        },

      });

    };



    // Remote track handler

    pc.ontrack = (e) => {

      const [stream] = e.streams;

      if (!stream) return;

      const map = new Map(this.state.remoteStreams);

      map.set(peerUserId, stream);

      this.patch({ remoteStreams: map });

    };



    // Connection state

    pc.onconnectionstatechange = async () => {

      const call = this.state.call;

      if (!call) return;

      if (pc.connectionState === 'connected') {

        this.patch({ status: 'connected' });

        this.socket.socket?.emit('call:mark-connected', { callId: call.id });

        this.startStatsLoop();

        this.startHeartbeat();

      } else if (pc.connectionState === 'failed') {

        // Attempt ICE restart.

        try {

          const offer = await pc.createOffer({ iceRestart: true });

          await pc.setLocalDescription(offer);

          this.socket.socket?.emit('call:offer', {

            callId: call.id,

            toUserId: peerUserId,

            sdp: offer.sdp ?? '',

          });

        } catch {

          this.patch({ status: 'ended' });

          this.cleanupLocal();

        }

      }

    };



    this.pcs.set(peerUserId, pc);

    return pc;

  }



  private closePeer(peerUserId: number): void {

    const pc = this.pcs.get(peerUserId);

    if (pc) {

      pc.close();

      this.pcs.delete(peerUserId);

    }

    const map = new Map(this.state.remoteStreams);

    map.delete(peerUserId);

    this.patch({ remoteStreams: map });

  }



  private async refreshIceServers(): Promise<void> {

    try {

      const creds: TurnCredentials = await this.api.getTurnCredentials();

      this.iceServers = [

        {

          urls: creds.urls,

          username: creds.username,

          credential: creds.credential,

        },

      ];

    } catch {

      this.iceServers = ICE_SERVERS_FALLBACK;

    }

  }



  private async getUserMedia(callType: CallType): Promise<MediaStream> {

    return navigator.mediaDevices.getUserMedia({

      audio: {

        echoCancellation: true,

        noiseSuppression: true,

        autoGainControl: true,

      },

        ? {

            width: { ideal: 1280 },

            height: { ideal: 720 },

            frameRate: { ideal: 30 },

          }

        : false,

    });

  }



  private broadcastMediaState(): void {

    const call = this.state.call;

    if (!call) return;

    this.socket.socket?.emit('call:media-state', {

      callId: call.id,

      micMuted: this.state.micMuted,

      cameraEnabled: this.state.cameraOn,

      screenSharing: this.state.screenSharing,

    });

  }



  private startStatsLoop(): void {

    if (this.statsTimer !== null) return;

    this.statsTimer = window.setInterval(async () => {

      const pc = [...this.pcs.values()][0];

      if (!pc) return;

      try {

        const stats = await pc.getStats();

        let rtt = 0, packetLoss = 0, inboundPackets = 0, lostPackets = 0;

        stats.forEach((r) => {

          if (r.type === 'candidate-pair' && (r as RTCIceCandidatePairStats).state === 'succeeded') {

            rtt = Math.max(rtt, Number((r as RTCIceCandidatePairStats).currentRoundTripTime ?? 0));

          }

          if (r.type === 'inbound-rtp') {

            const ir = r as RTCInboundRtpStreamStats;

            inboundPackets += Number(ir.packetsReceived ?? 0);

            lostPackets += Number(ir.packetsLost ?? 0);

          }

        });

        if (inboundPackets + lostPackets > 0) {

          packetLoss = lostPackets / (inboundPackets + lostPackets);

        }

        const level: 'excellent' | 'good' | 'poor' =

          rtt < 0.15 && packetLoss < 0.01 ? 'excellent' :

          rtt < 0.35 && packetLoss < 0.05 ? 'good' : 'poor';

        this.patch({ quality: { level, rtt, packetLoss } });

      } catch { /* ignore */ }

    }, 5_000);

  }



  private startHeartbeat(): void {

    if (this.heartbeatTimer !== null) return;

    this.heartbeatTimer = window.setInterval(() => {

      const call = this.state.call;

      if (!call) return;

      this.socket.socket?.emit('call:heartbeat', { callId: call.id });

    }, 15_000);

  }



  private cleanupLocal(): void {

    for (const [id] of this.pcs) this.closePeer(id);



    const ls = this.state.localStream;

    if (ls) {

      for (const t of ls.getTracks()) t.stop();

    }



    if (this.statsTimer !== null) {

      window.clearInterval(this.statsTimer);

      this.statsTimer = null;

    }

    if (this.heartbeatTimer !== null) {

      window.clearInterval(this.heartbeatTimer);

      this.heartbeatTimer = null;

    }



    this.patch({

      call: null,

      direction: null,

      localStream: null,

      remoteStreams: new Map(),

      status: 'idle',

      micMuted: false,

      cameraOn: true,

      screenSharing: false,

      quality: { level: 'excellent', rtt: 0, packetLoss: 0 },

    });

    this.pendingCandidates.clear();

  }



  private patch(patch: Partial<CallUiState>): void {

    this.state = { ...this.state, ...patch };

    this.listeners.forEach((l) => l(this.state));

  }

}

  v: 2;

    id: number;

const SALT = 'chat-widget-v2';
const STORAGE_PREFIX = 'chat_widget_session:';
const MARKER_PREFIX = 'chat_widget_active:';

function cryptoBytes(data: Uint8Array): Uint8Array<ArrayBuffer> {
  const bytes = new Uint8Array(new ArrayBuffer(data.byteLength));
  bytes.set(data);
  return bytes;
}

function storageKey(appId: string): string {
  const origin = typeof location !== 'undefined' ? location.origin : 'unknown';
  return `${STORAGE_PREFIX}${appId}:${origin}`;
}

function markerKey(appId: string): string {
  const origin = typeof location !== 'undefined' ? location.origin : 'unknown';
  return `${MARKER_PREFIX}${appId}:${origin}`;
}

async function deriveKey(appId: string): Promise<CryptoKey> {
  const enc = new TextEncoder();

  const keyMaterial = await crypto.subtle.importKey(
    'raw',
    cryptoBytes(enc.encode(`${SALT}:${appId}`)),
    { name: 'PBKDF2' },
    false,
    ['deriveKey'],
  );

  return crypto.subtle.deriveKey(
    {
      name: 'PBKDF2',
      salt: cryptoBytes(enc.encode(SALT)),
      iterations: 100_000,
      hash: 'SHA-256',
    },
    keyMaterial,
    { name: 'AES-GCM', length: 256 },
    false,
    ['encrypt', 'decrypt'],
  );
}

function toBase64(buf: ArrayBuffer | Uint8Array): string {
  const bytes = buf instanceof Uint8Array ? buf : new Uint8Array(buf);
  let s = '';
  for (let i = 0; i < bytes.byteLength; i++) s += String.fromCharCode(bytes[i]!);
  return btoa(s);
}

function fromBase64(s: string): Uint8Array<ArrayBuffer> {
  const bin = atob(s);
  const bytes = new Uint8Array(new ArrayBuffer(bin.length));

  for (let i = 0; i < bin.length; i++) {
    bytes[i] = bin.charCodeAt(i);
  }

  return bytes;
}

async function encrypt(appId: string, data: string): Promise<string> {
  const key = await deriveKey(appId);

  const iv = crypto.getRandomValues(
    new Uint8Array(new ArrayBuffer(12)),
  );

  const encoded = cryptoBytes(new TextEncoder().encode(data));

  const cipher = await crypto.subtle.encrypt(
    { name: 'AES-GCM', iv },
    key,
    encoded,
  );

  return `${toBase64(iv)}.${toBase64(cipher)}`;
}

async function decrypt(appId: string, blob: string): Promise<string> {
  const [ivB64, ctB64] = blob.split('.');

  if (!ivB64 || !ctB64) {
    throw new Error('invalid blob');
  }

  const key = await deriveKey(appId);

  const iv = fromBase64(ivB64);
  const ct = fromBase64(ctB64);

  const plain = await crypto.subtle.decrypt(
    { name: 'AES-GCM', iv },
    key,
    ct,
  );

  return new TextDecoder().decode(plain);
}

export const sessionStore = {
  async save(session: Omit<StoredSession, 'v' | 'savedAt'>): Promise<void> {
    try {
      const payload: StoredSession = { v: 2, savedAt: Date.now(), ...session };
      const blob = await encrypt(session.appId, JSON.stringify(payload));
      localStorage.setItem(storageKey(session.appId), blob);
      // Also drop a lightweight marker we can check without decryption
      localStorage.setItem(

  },

    try {
      const blob = localStorage.getItem(storageKey(appId));
      if (!blob) return null;
      const plain = await decrypt(appId, blob);
      const parsed = JSON.parse(plain) as StoredSession;
      if (parsed.v !== 2) return null;
      if (Date.now() - parsed.savedAt > 90 * 24 * 3600 * 1000) return null;
      return parsed;
    } catch (err) {

      console.warn('[ChatWidget] failed to load session', err);
      return null;
    }
  },

import { store } from '../state/store';
import type { Conversation, Message } from '../types';

interface NotificationPrefs {
  messageSound: boolean;
  messagePopup: boolean;
  callSound: boolean;
  callPopup: boolean;
}

const DEFAULT_PREFS: NotificationPrefs = {
  messageSound: true,
  messagePopup: true,
  callSound: true,
  callPopup: true,
};

class NotificationService {
  private prefs: NotificationPrefs = { ...DEFAULT_PREFS };
  private audioCtx: AudioContext | null = null;
  private granted = false;

  async init(): Promise<void> {
    if (typeof document !== 'undefined') {
      document.addEventListener('visibilitychange', () => { /* noop */ });
    }
  }

  async requestPermission(): Promise<boolean> {
    if (!('Notification' in window)) { this.granted = false; return false; }
    if (Notification.permission === 'granted') { this.granted = true; return true; }
    if (Notification.permission === 'denied') { this.granted = false; return false; }
    try {
      const result = await Notification.requestPermission();
      this.granted = result === 'granted';
      return this.granted;
    } catch { this.granted = false; return false; }
  }

  setPrefs(next: Partial<NotificationPrefs>): void {
    this.prefs = { ...this.prefs, ...next };
  }

  shouldNotify(conversation: Conversation | undefined, msg: Message): boolean {
    const me = store.get().me;
    if (!me) return false;
    if (msg.senderId === me.id) return false;
    if (!this.prefs.messagePopup && !this.prefs.messageSound) return false;
    const activeId = store.get().activeConversationId;
    const isActive = conversation?.id === activeId;
    const tabFocused = !document.hidden;
    if (isActive && tabFocused) return false;
    return true;
  }

  notifyMessage(conversation: Conversation | undefined, msg: Message): void {
    if (!this.shouldNotify(conversation, msg)) return;
    if (this.prefs.messageSound) this.playBeep();
    if (this.prefs.messagePopup && this.granted) {
      const title = this.titleFor(conversation, msg);
      const body = msg.content ?? this.fallbackBody(msg);
      this.showPopup(title, body, () => {
        window.focus();
        if (conversation) store.set({ isOpen: true, activeConversationId: conversation.id });
      });
    }
  }

  notifyCall(from: string, isVideo: boolean): void {
    if (this.prefs.callSound) this.playRingtone();
    if (this.prefs.callPopup && this.granted) {
      this.showPopup(`Incoming ${isVideo ? 'video' : 'audio'} call`, `From ${from}`, () => window.focus());
    }
  }

  private titleFor(conversation: Conversation | undefined, msg: Message): string {
    if (!conversation) return 'New message';
    if (conversation.type === 'group') return conversation.name ?? 'Group';
    const other = conversation.members.find((m) => m.userId === msg.senderId);
    return other ? `User #${other.userId}` : 'New message';
  }

  private fallbackBody(msg: Message): string {
    switch (msg.type) {
      case 'image': return '📷 Photo';
      case 'video': return '🎥 Video';
      case 'audio': return '🎵 Audio';
      case 'file': return '📄 File';
      case 'voice': return '🎤 Voice message';
      default: return 'New message';
    }
  }

  private showPopup(title: string, body: string, onClick: () => void): void {
    try {
      const n = new Notification(title, { body, icon: '/favicon.ico', tag: 'chat-widget-' + Date.now(), silent: true });
      n.onclick = () => { onClick(); n.close(); };
    } catch { /* noop */ }
  }

  private ensureAudioCtx(): AudioContext | null {
    if (typeof window === 'undefined') return null;
    const W = window as unknown as { AudioContext?: typeof AudioContext; webkitAudioContext?: typeof AudioContext };
    const Ctor = W.AudioContext || W.webkitAudioContext;
    if (!Ctor) return null;
    if (!this.audioCtx) this.audioCtx = new Ctor();
    return this.audioCtx;
  }

  private playBeep(): void {
    const ctx = this.ensureAudioCtx();
    if (!ctx) return;
    try {
      if (ctx.state === 'suspended') void ctx.resume();
      const now = ctx.currentTime;
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.type = 'sine';
      osc.frequency.setValueAtTime(880, now);
      osc.frequency.exponentialRampToValueAtTime(1320, now + 0.08);
      gain.gain.setValueAtTime(0.0001, now);
      gain.gain.exponentialRampToValueAtTime(0.15, now + 0.02);
      gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.25);
      osc.connect(gain).connect(ctx.destination);
      osc.start(now);
      osc.stop(now + 0.3);
    } catch { /* noop */ }
  }

  private playRingtone(): void {
    const ctx = this.ensureAudioCtx();
    if (!ctx) return;
    try {
      if (ctx.state === 'suspended') void ctx.resume();
      const now = ctx.currentTime;
      for (let i = 0; i < 3; i++) {
        const t0 = now + i * 0.6;
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(660, t0);
        osc.frequency.setValueAtTime(880, t0 + 0.15);
        gain.gain.setValueAtTime(0.0001, t0);
        gain.gain.exponentialRampToValueAtTime(0.12, t0 + 0.02);
        gain.gain.exponentialRampToValueAtTime(0.0001, t0 + 0.35);
        osc.connect(gain).connect(ctx.destination);
        osc.start(t0);
        osc.stop(t0 + 0.4);
      }
    } catch { /* noop */ }
  }
}

export interface Recording {
  blob: Blob;
  mimeType: string;
  durationMs: number;
}

export class VoiceRecorder {
  private mediaRecorder: MediaRecorder | null = null;
  private chunks: BlobPart[] = [];
  private startedAt = 0;
  private stopResolvers: Array<(r: Recording) => void> = [];

  async start(): Promise<void> {
    if (this.mediaRecorder) throw new Error('Already recording');
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    const mimeCandidates = ['audio/webm;codecs=opus', 'audio/webm', 'audio/ogg;codecs=opus', 'audio/mp4'];
    const mimeType = mimeCandidates.find((m) => MediaRecorder.isTypeSupported(m)) ?? '';

    this.chunks = [];
    this.startedAt = performance.now();
    this.mediaRecorder = new MediaRecorder(stream, mimeType ? { mimeType } : undefined);

    this.mediaRecorder.ondataavailable = (e) => { if (e.data && e.data.size > 0) this.chunks.push(e.data); };
    this.mediaRecorder.onstop = () => {
      const durationMs = Math.round(performance.now() - this.startedAt);
      const blob = new Blob(this.chunks, { type: this.mediaRecorder?.mimeType || 'audio/webm' });
      const result: Recording = { blob, mimeType: blob.type || 'audio/webm', durationMs };
      this.stopResolvers.forEach((r) => r(result));
      this.stopResolvers = [];
      stream.getTracks().forEach((t) => t.stop());
      this.mediaRecorder = null;
    };

    this.mediaRecorder.start();
  }

  async stop(): Promise<Recording> {
    if (!this.mediaRecorder) throw new Error('Not recording');
    return new Promise<Recording>((resolve) => {
      this.stopResolvers.push(resolve);
      this.mediaRecorder!.stop();
    });
  }

  cancel(): void {
    if (!this.mediaRecorder) return;
    this.stopResolvers = [];
    this.mediaRecorder.onstop = null;
    try { this.mediaRecorder.stop(); } catch { /* ignore */ }
    this.mediaRecorder = null;
    this.chunks = [];
  }

  isRecording(): boolean { return this.mediaRecorder !== null; }
}

    // ---------------------------------------------------------------

          email: stored.user.email,

      sessionStore.clear(config.appId);

        // eslint-disable-next-line no-console
        console.warn('[ChatWidget] auto-login failed, falling back to form', msg);
        // Fall through to widget-token / login-form handling.
      }
    }

    // ---------------------------------------------------------------
    if (config.token) {

        );

        await postAuthSetup(rt);

        }

      }
    } else if (config.allowLogin !== false) {
      store.setNeedsAuth(
        initialResetToken ? 'Reset your password' : 'No token provided',
      );
    } else {

import type { AuthStatus, AuthTokens, Conversation, CurrentUser, Message } from '../types';

export interface WidgetState {
  authStatus: AuthStatus;
  accessToken: string | null;
  refreshToken: string | null;
  me: CurrentUser | null;
  authError: string | null;

  isOpen: boolean;
  isLoading: boolean;
  error: string | null;
  theme: 'light' | 'dark';

  conversations: Conversation[];
  activeConversationId: number | null;
  messagesByConversation: Record<number, Message[]>;
  typingByConversation: Record<number, Set<number>>;
  onlineUsers: Set<number>;

  historyCursorByConversation: Record<number, string | null>;
  historyHasMoreByConversation: Record<number, boolean>;
  historyLoadingByConversation: Record<number, boolean>;
  unreadByConversation: Record<number, number>;
}

type Listener = (state: WidgetState) => void;

const initialState: WidgetState = {
  authStatus: 'idle',
  accessToken: null,
  refreshToken: null,
  me: null,
  authError: null,
  isOpen: false,
  isLoading: false,
  error: null,
  theme: 'light',
  conversations: [],
  activeConversationId: null,
  messagesByConversation: {},
  typingByConversation: {},
  onlineUsers: new Set(),
  historyCursorByConversation: {},
  historyHasMoreByConversation: {},
  historyLoadingByConversation: {},
  unreadByConversation: {},
};

class Store {
  private state: WidgetState = { ...initialState };
  private listeners = new Set<Listener>();

  get(): WidgetState { return this.state; }

  set(patch: Partial<WidgetState> | ((s: WidgetState) => Partial<WidgetState>)): void {
    const next = typeof patch === 'function' ? patch(this.state) : patch;
    this.state = { ...this.state, ...next };
    this.listeners.forEach((l) => l(this.state));
  }

  subscribe(listener: Listener): () => void {
    this.listeners.add(listener);
    return () => this.listeners.delete(listener);
  }

    /**
   * Wipe conversational state (used when switching users),
   * but DO NOT touch auth tokens unless explicitly asked.
   */
  resetConversationState(): void {
    this.set({
      conversations: [],
      activeConversationId: null,
      messagesByConversation: {},
      typingByConversation: {},
      onlineUsers: new Set(),
      historyCursorByConversation: {},
      historyHasMoreByConversation: {},
      historyLoadingByConversation: {},
      unreadByConversation: {},
      isLoading: false,
      error: null,
    });
  }

  /**
   * Full reset — used only on explicit logout.
   */
  reset(): void {
    this.state = { ...initialState, onlineUsers: new Set() };
    this.listeners.forEach((l) => l(this.state));
  }

  setAuth(tokens: AuthTokens): void {
    this.set({
      authStatus: 'authenticated',
      authError: null,
      accessToken: tokens.accessToken,
      refreshToken: tokens.refreshToken,
      me: tokens.user,
    });
  }

  setNeedsAuth(reason?: string): void {
    this.set({
      authStatus: 'needs-auth',
      authError: reason ?? null,
      accessToken: null,
      refreshToken: null,
      me: null,
    });
  }

  upsertConversation(conv: Conversation): void {
    this.set((s) => {
      const existing = s.conversations.findIndex((c) => c.id === conv.id);
      const conversations =
        existing === -1
          ? [conv, ...s.conversations]
          : s.conversations.map((c) => (c.id === conv.id ? { ...c, ...conv } : c));
      conversations.sort((a, b) => {
        const at = a.lastMessageAt ? Date.parse(a.lastMessageAt) : 0;
        const bt = b.lastMessageAt ? Date.parse(b.lastMessageAt) : 0;
        return bt - at;
      });
      return { conversations };
    });
  }

  addMessage(msg: Message): void {
    this.set((s) => {
      const list = s.messagesByConversation[msg.conversationId] ?? [];
      if (list.some((m) => m.id === msg.id)) return {};
      const next = [...list, msg].sort((a, b) => a.id - b.id);

      const unread = { ...s.unreadByConversation };
      const isActive = s.activeConversationId === msg.conversationId;
      const isMine = s.me?.id === msg.senderId;
      if (!isActive && !isMine) {
        unread[msg.conversationId] = (unread[msg.conversationId] ?? 0) + 1;
      }

      return {
        messagesByConversation: { ...s.messagesByConversation, [msg.conversationId]: next },
        unreadByConversation: unread,
      };
    });
  }

  prependMessages(conversationId: number, msgs: Message[]): void {
    this.set((s) => {
      const existing = s.messagesByConversation[conversationId] ?? [];
      const known = new Set(existing.map((m) => m.id));
      const merged = [...msgs.filter((m) => !known.has(m.id)), ...existing].sort((a, b) => a.id - b.id);
      return { messagesByConversation: { ...s.messagesByConversation, [conversationId]: merged } };
    });
  }

  setMessages(conversationId: number, msgs: Message[]): void {
    this.set((s) => ({
      messagesByConversation: { ...s.messagesByConversation, [conversationId]: [...msgs].sort((a, b) => a.id - b.id) },
    }));
  }

  updateMessage(conversationId: number, msg: Message): void {
    this.set((s) => {
      const list = s.messagesByConversation[conversationId] ?? [];
      const next = list.map((m) => (m.id === msg.id ? { ...m, ...msg } : m));
      return { messagesByConversation: { ...s.messagesByConversation, [conversationId]: next } };
    });
  }

  setTyping(conversationId: number, userId: number, isTyping: boolean): void {
    this.set((s) => {
      const set = new Set(s.typingByConversation[conversationId] ?? []);
      if (isTyping) set.add(userId); else set.delete(userId);
      return { typingByConversation: { ...s.typingByConversation, [conversationId]: set } };
    });
  }

  setOnline(userId: number, online: boolean): void {
    this.set((s) => {
      const next = new Set(s.onlineUsers);
      if (online) next.add(userId); else next.delete(userId);
      return { onlineUsers: next };
    });
  }

  clearUnread(conversationId: number): void {
    this.set((s) => {
      const unread = { ...s.unreadByConversation };
      delete unread[conversationId];
      return { unreadByConversation: unread };

/**
 * Format a UTC timestamp into the current viewer's local time.
 * Uses Intl.DateTimeFormat with the browser's resolved timezone
 * so recipients in other regions see their own local clock.
 */
export function formatLocalTime(iso: string): string {
  const d = new Date(iso);
  return new Intl.DateTimeFormat(undefined, {
    hour: '2-digit',
    minute: '2-digit',
    hour12: undefined,
  }).format(d);
}

export function formatLocalDateTime(iso: string): string {
  const d = new Date(iso);
  const now = new Date();
  const sameDay =
    d.getFullYear() === now.getFullYear() &&
    d.getMonth() === now.getMonth() &&
    d.getDate() === now.getDate();

  const yesterday = new Date(now);
  yesterday.setDate(now.getDate() - 1);
  const isYesterday =
    d.getFullYear() === yesterday.getFullYear() &&
    d.getMonth() === yesterday.getMonth() &&
    d.getDate() === yesterday.getDate();

  const time = new Intl.DateTimeFormat(undefined, {
    hour: '2-digit',
    minute: '2-digit',
  }).format(d);

  if (sameDay) return time;
  if (isYesterday) return `Yesterday ${time}`;

  const date = new Intl.DateTimeFormat(undefined, {
    month: 'short',
    day: 'numeric',
  }).format(d);
  return `${date} ${time}`;
}

/** Format duration in seconds → "m:ss" */
export function formatDuration(seconds: number): string {
  const m = Math.floor(seconds / 60);
  const s = seconds % 60;
  return `${m}:${String(s).padStart(2, '0')}`;
}

/** Human-size helper */
export function humanSize(bytes: number): string {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
}

  replyToId: z.coerce.number().int().positive().nullable().optional(),

  socket: Socket,
  ctx: SocketContext,

  const convos = new ConversationsRepository(app.db);

  const memberships: Array<{ conversation_id: number }> = await app
    .db('conversation_members')
    .where({ user_id: ctx.userId })
    .whereNull('left_at')
    .select('conversation_id');

  for (const m of memberships) socket.join(`t:${ctx.tenantId}:c:${m.conversation_id}`);

    socket.on('conversation:created', async (payload, ack) => {

        .object({ conversationId: z.coerce.number().int().positive() })
        .parse(payload);
      const member = await convos.getMember(conversationId, ctx.userId);
      if (!member) return ack?.({ error: 'NOT_A_MEMBER' });

      ack?.({ ok: true });
    } catch {
      ack?.({ error: 'INVALID_PAYLOAD' });
    }
  });

      const member = await convos.getMember(conversationId, ctx.userId);
      if (!member) return ack?.({ error: 'NOT_A_MEMBER' });

      ack?.({ ok: true });
    } catch { ack?.({ error: 'INVALID_PAYLOAD' }); }
  });

    try {
      const { conversationId } = z.object({ conversationId: z.coerce.number().int().positive() }).parse(payload);
      socket.leave(`t:${ctx.tenantId}:c:${conversationId}`);
      ack?.({ ok: true });

        type: parsed.type,
        content: parsed.content ?? null,
        replyToId: parsed.replyToId ?? null,
        attachmentIds: parsed.attachmentIds ?? [],
      });
      if (parsed.clientId) dto.clientId = parsed.clientId;
      const room = `t:${ctx.tenantId}:c:${parsed.conversationId}`;
      socket.to(room).emit('message:new', dto);
      socket.emit('message:new', dto);
      ack?.({ ok: true, message: dto });
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      ack?.({ error: 'SEND_FAILED', message });
      socket.emit('error', { code: 'SEND_FAILED', message });
    }
  });

        messageId: z.coerce.number().int().positive(),
        content: z.string().min(1).max(8000),

      ack?.({ error: 'EDIT_FAILED', message });
    }

      const parsed = z.object({ messageId: z.coerce.number().int().positive() }).parse(payload);
      const row = await service.loadByIdOrThrow(ctx.tenantId, parsed.messageId);
      const conversationId = row.conversation_id;
      await service.delete(ctx.tenantId, ctx.userId, parsed.messageId);
      const out = { messageId: parsed.messageId, conversationId };
      const room = `t:${ctx.tenantId}:c:${conversationId}`;
      socket.to(room).emit('message:deleted', out);
      socket.emit('message:deleted', out);
      ack?.({ ok: true });
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      ack?.({ error: 'DELETE_FAILED', message });

  socket.on('message:react', async (payload, ack) => {
    try {
      const parsed = z.object({
        messageId: z.coerce.number().int().positive(),
        reaction: z.string().min(1).max(16),
        added: z.boolean().default(true),

      const room = `t:${ctx.tenantId}:c:${conversationId}`;

    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      ack?.({ error: 'READ_FAILED', message });
    }
  });
}

import type { FastifyInstance } from 'fastify';
import type { Socket } from 'socket.io';
import { z } from 'zod';

import type { SocketContext } from '../context.js';
import { getIO, tenantRoom, userRoom } from '../io.js';

const PRESENCE_TTL = 86_400;      // 24 hours
const LAST_SEEN_TTL = 2_592_000;  // 30 days

const PresenceGetSchema = z.object({
  userId: z.coerce.number().int().positive(),
});

function presenceKey(tenantId: number, userId: number): string {
  return `presence:${tenantId}:${userId}`;
}

function lastSeenKey(tenantId: number, userId: number): string {
  return `presence:last_seen:${tenantId}:${userId}`;
}

async function countActiveSockets(
  tenantId: number,
  userId: number,
): Promise<number> {
  try {
    const sockets = await getIO().in(userRoom(tenantId, userId)).fetchSockets();
    return sockets.length;
  } catch {
    return 0;
  }
}

async function userExistsInTenant(
  app: FastifyInstance,
  tenantId: number,
  userId: number,
): Promise<boolean> {
  const row = await app.db('users')
    .where({ tenant_id: tenantId, id: userId })
    .first<{ id: number }>();
  return Boolean(row);
}

interface PresencePayload {
  userId: number;
  online: boolean;
  lastSeen: string | null;
}

async function readPresence(
  app: FastifyInstance,
  tenantId: number,
  userId: number,
): Promise<PresencePayload> {
  const active = await countActiveSockets(tenantId, userId);
  if (active > 0) {
    return { userId, online: true, lastSeen: null };
  }

  const cached = await app.redis.get(presenceKey(tenantId, userId));
  if (cached === 'online') {
    return { userId, online: true, lastSeen: null };
  }

  let lastSeen = await app.redis.get(lastSeenKey(tenantId, userId));
  if (!lastSeen) {
    try {
      const row = await app.db('users')
        .where({ tenant_id: tenantId, id: userId })
        .first<{ last_seen_at?: Date | string | null }>();
      if (row?.last_seen_at) {
        lastSeen = new Date(row.last_seen_at).toISOString();
      }
    } catch {
      /* users.last_seen_at may not exist — ignore */
    }
  }

  return { userId, online: false, lastSeen: lastSeen ?? null };
}

export async function registerPresenceHandlers(
  app: FastifyInstance,
  socket: Socket,
  ctx: SocketContext,
): Promise<void> {
  const me = {
    tenantId: ctx.tenantId,
    userId: ctx.userId,
  };

  // ---- on connect ----
  const activeCount = await countActiveSockets(me.tenantId, me.userId);

  try {
    await app.redis.set(presenceKey(me.tenantId, me.userId), 'online', 'EX', PRESENCE_TTL);
  } catch (err) {
    app.log.warn({ err }, 'presence: set online failed');
  }

  if (activeCount <= 1) {
    getIO().to(tenantRoom(me.tenantId)).emit('presence:online', {
      userId: me.userId,
      online: true,
      lastSeen: null,
    });
  }

  // ---- presence:get ----
  socket.on('presence:get', async (payload: unknown, ack?: (r: unknown) => void) => {
    try {
      const { userId } = PresenceGetSchema.parse(payload);
      const exists = await userExistsInTenant(app, me.tenantId, userId);
      if (!exists) {
        ack?.({ ok: false, error: { code: 'USER_NOT_FOUND', message: 'User not found' } });
        return;
      }
      const presence = await readPresence(app, me.tenantId, userId);
      ack?.({ ok: true, data: presence });
    } catch (err) {
      const message = err instanceof Error ? err.message : 'Invalid presence request';
      ack?.({ ok: false, error: { code: 'PRESENCE_ERROR', message } });
    }
  });

  // ---- presence:heartbeat ----
  socket.on('presence:heartbeat', async () => {
    try {
      await app.redis.set(presenceKey(me.tenantId, me.userId), 'online', 'EX', PRESENCE_TTL);
    } catch {
      /* ignore */
    }
  });

  // ---- on disconnect ----
  socket.on('disconnect', async () => {
    // Wait one tick so the socket leaves the room before we count.
    await new Promise<void>((r) => setTimeout(r, 0));

    const remaining = await countActiveSockets(me.tenantId, me.userId);
    if (remaining > 0) {
      try {
        await app.redis.set(presenceKey(me.tenantId, me.userId), 'online', 'EX', PRESENCE_TTL);
      } catch {
        /* ignore */
      }
      return;
    }

    const lastSeen = new Date().toISOString();
    try {
      await app.redis.del(presenceKey(me.tenantId, me.userId));
      await app.redis.set(lastSeenKey(me.tenantId, me.userId), lastSeen, 'EX', LAST_SEEN_TTL);
    } catch {
      /* ignore */
    }

    getIO().to(tenantRoom(me.tenantId)).emit('presence:offline', {
      userId: me.userId,
      online: false,
      lastSeen,
    });
  });
}

export default registerPresenceHandlers;

      const { callId } = CallIdSchema.parse(payload);

      await service.cancel({ tenantId: ctx.tenantId, callId, userId: ctx.userId });

      const io = getIO();
      for (const p of call.participants) {
        io.to(userRoom(ctx.tenantId, p.userId)).emit('call:cancelled', {
          callId,
          byUserId: ctx.userId,
        });
      }
      ack?.({ ok: true });
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      ack?.({ error: 'CANCEL_FAILED', message });
    }
  });

  // -------- End --------
  socket.on('call:end', async (payload, ack) => {
    try {
      const parsed = z.object({
        callId: z.coerce.number().int().positive(),
        reason: z.enum(['completed', 'remote_ended', 'network_error']).optional(),
      }).parse(payload);

      const call = await service.getById(ctx.tenantId, parsed.callId);
      await service.end({
        tenantId: ctx.tenantId,
        callId: parsed.callId,
        userId: ctx.userId,
        reason: parsed.reason ?? 'completed',
      });

      const io = getIO();
      for (const p of call.participants) {
        io.to(userRoom(ctx.tenantId, p.userId)).emit('call:ended', {
          callId: parsed.callId,
          byUserId: ctx.userId,
          reason: parsed.reason ?? 'completed',
        });
      }
      ack?.({ ok: true });
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      ack?.({ error: 'END_FAILED', message });
    }
  });

  // -------- Offer --------
  socket.on('call:offer', async (payload, ack) => {
    try {
      const parsed = OfferSchema.parse(payload);
      const call = await service.getById(ctx.tenantId, parsed.callId);
      const ids = new Set(call.participants.map((p) => p.userId));
      if (!ids.has(ctx.userId) || !ids.has(parsed.toUserId)) {
        throw new Error('Not a participant');
      }
      getIO().to(userRoom(ctx.tenantId, parsed.toUserId)).emit('call:offer', {
        callId: parsed.callId,
        fromUserId: ctx.userId,
        sdp: parsed.sdp,
      });
      ack?.({ ok: true });
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      ack?.({ error: 'OFFER_FAILED', message });
    }
  });

  // -------- Answer --------
  socket.on('call:answer', async (payload, ack) => {
    try {
      const parsed = AnswerSchema.parse(payload);
      getIO().to(userRoom(ctx.tenantId, parsed.toUserId)).emit('call:answer', {
        callId: parsed.callId,
        fromUserId: ctx.userId,
        sdp: parsed.sdp,
      });
      ack?.({ ok: true });
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      ack?.({ error: 'ANSWER_FAILED', message });
    }
  });

  // -------- ICE candidate --------
  socket.on('call:ice', async (payload, ack) => {
    try {
      const parsed = IceSchema.parse(payload);
      getIO().to(userRoom(ctx.tenantId, parsed.toUserId)).emit('call:ice', {
        callId: parsed.callId,
        fromUserId: ctx.userId,
        candidate: parsed.candidate,
      });
      ack?.({ ok: true });
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      ack?.({ error: 'ICE_FAILED', message });
    }
  });

  // -------- Media state --------
  socket.on('call:media-state', async (payload, ack) => {
    try {
      const parsed = MediaStateSchema.parse(payload);
      const call = await service.getById(ctx.tenantId, parsed.callId);

      const patch: Record<string, 0 | 1> = {};
      if (typeof parsed.micMuted === 'boolean') patch.mic_muted = parsed.micMuted ? 1 : 0;
      if (typeof parsed.cameraEnabled === 'boolean') patch.camera_enabled = parsed.cameraEnabled ? 1 : 0;
      if (typeof parsed.screenSharing === 'boolean') patch.screen_sharing = parsed.screenSharing ? 1 : 0;

      if (Object.keys(patch).length > 0) {
        await app.db('call_participants')
          .where({ call_id: parsed.callId, user_id: ctx.userId })
          .update({ ...patch, updated_at: app.db.fn.now() });
      }

      const io = getIO();
      for (const p of call.participants) {
        if (p.userId === ctx.userId) continue;
        io.to(userRoom(ctx.tenantId, p.userId)).emit('call:media-state', {
          callId: parsed.callId,
          userId: ctx.userId,
          micMuted: parsed.micMuted,
          cameraEnabled: parsed.cameraEnabled,
          screenSharing: parsed.screenSharing,
        });
      }
      ack?.({ ok: true });
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      ack?.({ error: 'MEDIA_STATE_FAILED', message });
    }
  });

  // -------- Heartbeat --------
  socket.on('call:heartbeat', async (payload, ack) => {
    try {
      const { callId } = CallIdSchema.parse(payload);
      await app.redis.set(`call:${callId}:heartbeat:${ctx.userId}`, '1', 'EX', 30);
      ack?.({ ok: true });
    } catch {
      ack?.({ error: 'HEARTBEAT_FAILED' });
    }
  });
}

import { Server as SocketIOServer, type Socket } from 'socket.io';
import { createAdapter } from '@socket.io/redis-adapter';
import type { FastifyInstance } from 'fastify';
import Redis from 'ioredis';

import { env } from '../config/env.js';
import { TokenService } from '../modules/auth/token.service.js';
import type { SocketContext } from './context.js';
import { registerHandlers } from './register.js';

let io: SocketIOServer | null = null;

export function getIO(): SocketIOServer {
  if (!io) throw new Error('Socket.IO not initialized');
  return io;
}

/**
 * Room name helpers — used by presence, calls, messages.
 * Keep these here so every handler uses the same convention.
 */
export function tenantRoom(tenantId: number): string {
  return `t:${tenantId}`;
}

export function userRoom(tenantId: number, userId: number): string {
  return `t:${tenantId}:u:${userId}`;
}

export function conversationRoom(tenantId: number, conversationId: number): string {
  return `t:${tenantId}:c:${conversationId}`;
}

export async function initSocketIO(app: FastifyInstance): Promise<SocketIOServer> {
  const server = new SocketIOServer(app.server, {
    path: '/socket.io',
    cors: {
      origin: true,
      credentials: true,
    },
    transports: ['websocket', 'polling'],
    pingInterval: 25_000,
    pingTimeout: 20_000,
    maxHttpBufferSize: 1_000_000,
  });

  // Redis adapter (multi-instance broadcasting)
  const pubClient = new Redis({
    host: env.REDIS_HOST,
    port: env.REDIS_PORT,
    password: env.REDIS_PASSWORD || undefined,
    db: env.REDIS_DB,
    keyPrefix: env.REDIS_KEY_PREFIX,
  });
  const subClient = pubClient.duplicate();
  server.adapter(createAdapter(pubClient, subClient));

  const tokens = new TokenService(app.db, app.redis);

  // ---- Auth handshake ----
  server.use(async (socket, next) => {
    try {
      const auth = (socket.handshake.auth || {}) as { token?: string; deviceId?: string };
      const headerAuth = socket.handshake.headers.authorization;
      const bearer =
        auth.token ||
        (typeof headerAuth === 'string' && headerAuth.startsWith('Bearer ')
          ? headerAuth.slice(7).trim()
          : undefined);

      if (!bearer) return next(new Error('UNAUTHORIZED'));

      const claims = await tokens.verifyAccessToken(bearer);
      const ctx: SocketContext = {
        userId: Number(claims.sub),
        tenantId: Number(claims.tenantId),
        role: String(claims.role),
        deviceId: auth.deviceId ?? 'web',
      };
      socket.data.ctx = ctx;
      next();
    } catch (err) {
      app.log.warn({ err }, 'socket auth failed');
      next(new Error('UNAUTHORIZED'));
    }
  });

  // ---- On connection ----
  server.on('connection', (socket: Socket) => {
    const ctx = socket.data.ctx as SocketContext;

    socket.join(tenantRoom(ctx.tenantId));
    socket.join(userRoom(ctx.tenantId, ctx.userId));

    app.log.info(
      { socketId: socket.id, userId: ctx.userId, tenantId: ctx.tenantId },
      'socket connected',
    );

    void registerHandlers(app, socket);

    socket.on('disconnect', (reason) => {
      app.log.info({ socketId: socket.id, reason }, 'socket disconnected');
    });
  });

  app.addHook('onClose', async () => {
    await server.close();
    await pubClient.quit();
    await subClient.quit();
    app.log.info('socket.io closed');
  });

  io = server;
  return server;
}

import type { Knex } from 'knex';
import { randomUUID } from 'node:crypto';

import { ForbiddenError, NotFoundError } from '../../utils/errors.js';
import { buildPage, decodeCursor, type CursorPage } from '../../utils/pagination.js';
import { ConversationsRepository } from '../conversations/conversations.repository.js';
import { MessagesRepository, type MessageRow } from './messages.repository.js';
import { AttachmentsService } from '../attachments/attachments.service.js';
import type { MessageDTO, MessageType } from '@chat/shared-types';

export interface CreateMessageInput {
  tenantId: number;
  conversationId: number;
  senderId: number;
  type?: MessageType;
  content?: string | null;
  replyToId?: number | null;
  attachmentIds?: number[];
}

export class MessagesService {
  private readonly repo: MessagesRepository;
  private readonly convos: ConversationsRepository;
  private readonly attachments: AttachmentsService;

  constructor(private readonly db: Knex) {
    this.repo = new MessagesRepository(db);
    this.convos = new ConversationsRepository(db);
    this.attachments = new AttachmentsService(db);
  }

  async create(input: CreateMessageInput): Promise<MessageDTO> {
    const member = await this.convos.getMember(input.conversationId, input.senderId);
    if (!member) throw new ForbiddenError('Not a member of this conversation');

    const type: MessageType = input.type ?? 'text';
    const content = (input.content ?? '').toString().slice(0, 8000);
    const attachmentIds = input.attachmentIds ?? [];

    if (type === 'text' && content.trim().length === 0 && attachmentIds.length === 0) {
      throw new ForbiddenError('Empty message');
    }
    if (attachmentIds.length > 5) throw new ForbiddenError('Too many attachments');

    if (input.replyToId) {
      const parent = await this.repo.findById(input.tenantId, input.replyToId);
      if (!parent || parent.conversation_id !== input.conversationId) {
        throw new NotFoundError('Reply target not found in this conversation');
      }
    }

    const messageUuid = randomUUID();
    const tenantId = input.tenantId;
    const conversationId = input.conversationId;
    const senderId = input.senderId;
    const replyToId = input.replyToId ?? null;

    const newMessageId: number = await this.db.transaction(async (trx) => {
      const insertedId = await this.repo.insert(trx, {
        uuid: messageUuid,
        tenantId,
        conversationId,
        senderId,
        type,
        content: content.length > 0 ? content : null,
        replyToId,
      });

      if (attachmentIds.length > 0) {
        await this.attachments.attachToMessage(trx, tenantId, insertedId, attachmentIds);
      }

      await trx('conversations')
        .where({ tenant_id: tenantId, id: conversationId })
        .update({ last_message_at: trx.fn.now(), updated_at: trx.fn.now() });

      await this.repo.upsertStatus(trx, insertedId, senderId, 'read');

      const members: Array<{ user_id: number }> = await trx('conversation_members')
        .where({ conversation_id: conversationId })
        .whereNull('left_at')
        .select('user_id');
      for (const m of members) {
        if (m.user_id === senderId) continue;
        await this.repo.upsertStatus(trx, insertedId, m.user_id, 'sent');
      }
      return insertedId;
    });

    const row = await this.repo.findById(tenantId, newMessageId);
    if (!row) throw new Error('Message vanished after insert');
    const attachMap = await this.attachments.loadForMessages(tenantId, [newMessageId]);
    return this.toDTO(row, undefined, attachMap.get(newMessageId) ?? []);
  }

    async history(
    tenantId: number,
    userId: number,
    conversationId: number,
    limit = 30,
    cursor?: string,
  ): Promise<CursorPage<MessageDTO>> {
    // Any member who has *ever* been in the conversation (even left)
    // can read messages sent while they were a member.
    // New members added later can read history from the conversation's start.
    // If you want to restrict new members to only messages after they joined,
    // replace getMember with getMemberHistory and filter by `joined_at`.
    const member = await this.convos.getMember(conversationId, userId);
    const historical = member ?? (await this.convos.getMemberHistory(conversationId, userId));
    if (!historical) throw new ForbiddenError('Not a member');

    const cursorId = cursor ? Number(decodeCursor(cursor)) : null;
    const rows = await this.repo.listByConversation(tenantId, conversationId, limit, cursorId);
    const page = buildPage(rows, limit);

    const ids = page.items.map((r) => r.id);
    const reactions = await this.repo.reactionsFor(ids);
    const attachMap = await this.attachments.loadForMessages(tenantId, ids);

    const reactionMap = new Map<number, MessageDTO['reactions']>();
    for (const r of reactions) {
      const arr = reactionMap.get(r.message_id) ?? [];
      arr.push({
        reaction: r.reaction,
        userId: r.user_id,
        createdAt: new Date(r.created_at).toISOString(),
      });
      reactionMap.set(r.message_id, arr);
    }

    return {
      ...page,
      items: page.items.map((r) =>
        this.toDTO(r, reactionMap.get(r.id), attachMap.get(r.id)),
      ),
    };
  }

  async search(
    tenantId: number,
    userId: number,
    conversationId: number,
    term: string,
    limit = 30,
  ): Promise<MessageDTO[]> {
    const member = await this.convos.getMember(conversationId, userId);
    if (!member) throw new ForbiddenError('Not a member');
    const rows = await this.repo.search(tenantId, conversationId, term, Math.min(limit, 50));
    return rows.map((r) => this.toDTO(r));
  }

  async edit(
    tenantId: number,
    userId: number,
    messageId: number,
    content: string,
  ): Promise<MessageDTO> {
    const row = await this.repo.findById(tenantId, messageId);
    if (!row) throw new NotFoundError('Message not found');
    if (row.sender_id !== userId) throw new ForbiddenError('Only the sender can edit');
    if (row.deleted_at) throw new ForbiddenError('Cannot edit a deleted message');

    const cleaned = content.slice(0, 8000);
    await this.db.transaction(async (trx) => {
      await this.repo.updateContent(trx, tenantId, messageId, cleaned);
    });
    const fresh = await this.repo.findById(tenantId, messageId);
    if (!fresh) throw new NotFoundError('Message not found after update');
    return this.toDTO(fresh);
  }

  async delete(tenantId: number, userId: number, messageId: number): Promise<void> {
    const row = await this.repo.findById(tenantId, messageId);
    if (!row) throw new NotFoundError('Message not found');

    const member = await this.convos.getMember(row.conversation_id, userId);
    if (!member) throw new ForbiddenError('Not a member');

    const isSender = row.sender_id === userId;
    const isGroupModerator = member.role === 'owner' || member.role === 'admin';
    if (!isSender && !isGroupModerator) {
      throw new ForbiddenError('Not allowed to delete this message');
    }

    await this.db.transaction(async (trx) => {
      await this.repo.softDelete(trx, tenantId, messageId, userId);
    });
  }

  async react(
    tenantId: number,
    userId: number,
    messageId: number,
    reaction: string,
  ): Promise<void> {
    const row = await this.repo.findById(tenantId, messageId);
    if (!row) throw new NotFoundError('Message not found');
    const member = await this.convos.getMember(row.conversation_id, userId);
    if (!member) throw new ForbiddenError('Not a member');

    const trimmed = reaction.trim().slice(0, 16);
    if (!trimmed) throw new ForbiddenError('Empty reaction');

    await this.db.transaction(async (trx) => {
      await this.repo.addReaction(trx, messageId, userId, trimmed);
    });
  }

  async unreact(
    tenantId: number,
    userId: number,
    messageId: number,
    reaction: string,
  ): Promise<void> {
    const row = await this.repo.findById(tenantId, messageId);
    if (!row) throw new NotFoundError('Message not found');
    const member = await this.convos.getMember(row.conversation_id, userId);
    if (!member) throw new ForbiddenError('Not a member');

    await this.db.transaction(async (trx) => {
      await this.repo.removeReaction(trx, messageId, userId, reaction.trim().slice(0, 16));
    });
  }

    async markRead(
    tenantId: number,
    userId: number,
    conversationId: number,
    messageIds: number[],
  ): Promise<number[]> {
    // Allow both current and historical members (users who left still
    // have their membership row). If neither exists, silently return [] —
    // this prevents stale sockets from throwing 403s into the widget.
    const active = await this.convos.getMember(conversationId, userId);
    const historical = active ?? (await this.convos.getMemberHistory(conversationId, userId));
    if (!historical) return [];

    const rows = await this.repo.findByIds(tenantId, messageIds);
    const validIds = rows
      .filter((r) => r.conversation_id === conversationId)
      .map((r) => r.id);
    if (validIds.length === 0) return [];

    await this.db.transaction(async (trx) => {
      await this.repo.markRead(trx, validIds, userId);
      // Only update last_read_at for members who haven't left.
      await trx('conversation_members')
        .where({ conversation_id: conversationId, user_id: userId })
        .whereNull('left_at')
        .update({ last_read_at: trx.fn.now() });
    });
    return validIds;
  }

  async loadByIdOrThrow(tenantId: number, messageId: number): Promise<MessageRow> {
    const row = await this.repo.findById(tenantId, messageId);
    if (!row) throw new NotFoundError('Message not found');
    return row;
  }

    toDTO(row: MessageRow, reactions?: MessageDTO['reactions'], attachments?: MessageDTO['attachments']): MessageDTO {
    return {
      id: row.id,
      uuid: row.uuid,
      conversationId: row.conversation_id,
      senderId: row.sender_id,
      type: row.type,
      content: row.deleted_at ? null : row.content,
      replyToId: row.reply_to_id,
      forwardedFromId: row.forwarded_from_id,
      editedAt: row.edited_at ? new Date(row.edited_at).toISOString() : null,
      deletedAt: row.deleted_at ? new Date(row.deleted_at).toISOString() : null,
      pinnedAt: row.pinned_at ? new Date(row.pinned_at).toISOString() : null,
      createdAt: new Date(row.created_at).toISOString(),
      reactions,
      attachments,
    };
  }

    async pin(
    tenantId: number,
    userId: number,
    messageId: number,
    pinned: boolean,
  ): Promise<MessageDTO> {
    const row = await this.repo.findById(tenantId, messageId);
    if (!row) throw new NotFoundError('Message not found');

    const member = await this.convos.getMember(row.conversation_id, userId);
    if (!member) throw new ForbiddenError('Not a member');

    await this.db.transaction(async (trx) => {
      await this.repo.setPinned(trx, tenantId, messageId, pinned ? userId : null);
    });

    const fresh = await this.repo.findById(tenantId, messageId);
    if (!fresh) throw new NotFoundError('Message not found after update');
    return this.toDTO(fresh);
  }

  async listPinned(tenantId: number, userId: number, conversationId: number): Promise<MessageDTO[]> {
    const member = await this.convos.getMember(conversationId, userId);
    if (!member) throw new ForbiddenError('Not a member');
    const rows = await this.repo.listPinned(tenantId, conversationId);
    return rows.map((r) => this.toDTO(r));
  }

  async forward(
    tenantId: number,
    userId: number,
    messageId: number,
    targetConversationIds: number[],
  ): Promise<MessageDTO[]> {
    const source = await this.repo.findById(tenantId, messageId);
    if (!source) throw new NotFoundError('Message not found');

    const sourceMember = await this.convos.getMember(source.conversation_id, userId);
    if (!sourceMember) throw new ForbiddenError('Not a member of source conversation');

    const created: MessageDTO[] = [];
    for (const targetId of targetConversationIds) {
      const targetMember = await this.convos.getMember(targetId, userId);
      if (!targetMember) continue; // skip silently for non-members

      const dto = await this.create({
        tenantId,
        conversationId: targetId,
        senderId: userId,
        type: source.type,
        content: source.content,
        replyToId: null,
        attachmentIds: [],
      });

      // Mark as forwarded
      await this.db('messages')
        .where({ tenant_id: tenantId, id: dto.id })
        .update({ forwarded_from_id: source.id });

      dto.forwardedFromId = source.id;
      created.push(dto);
    }
    return created;
  }
}

  ): Promise<MessageRow[]> {
    const q = this.db('messages')
      .where({ tenant_id: tenantId, conversation_id: conversationId })
      .orderBy('id', 'desc')
      .limit(limit + 1);
    if (cursorId) q.andWhere('id', '<', cursorId);
    return q;
  }

  async search(
    tenantId: number,
    conversationId: number,
    term: string,
    limit: number,
  ): Promise<MessageRow[]> {
    const like = `%${term.replace(/[%_]/g, (m) => '\\' + m)}%`;
    return this.db('messages')
      .where({ tenant_id: tenantId, conversation_id: conversationId })
      .whereNull('deleted_at')
      .andWhere('content', 'like', like)
      .orderBy('id', 'desc')
      .limit(limit);
  }

  async updateContent(
    trx: Knex.Transaction,
    tenantId: number,
    messageId: number,
    content: string,
  ): Promise<void> {
    await trx('messages')
      .where({ tenant_id: tenantId, id: messageId })
      .update({ content, edited_at: trx.fn.now() });
  }

  async softDelete(
    trx: Knex.Transaction,
    tenantId: number,
    messageId: number,
    byUserId: number,
  ): Promise<void> {
    await trx('messages')
      .where({ tenant_id: tenantId, id: messageId })
      .update({ deleted_at: trx.fn.now(), deleted_by: byUserId, content: null });
  }

  async addReaction(
    trx: Knex.Transaction,
    messageId: number,
    userId: number,
    reaction: string,
  ): Promise<void> {
    await trx('message_reactions')
      .insert({ message_id: messageId, user_id: userId, reaction })
      .onConflict(['message_id', 'user_id', 'reaction'])
      .ignore();
  }

  async removeReaction(
    trx: Knex.Transaction,
    messageId: number,
    userId: number,
    reaction: string,
  ): Promise<void> {
    await trx('message_reactions')
      .where({ message_id: messageId, user_id: userId, reaction })
      .delete();
  }

  async reactionsFor(messageIds: number[]): Promise<ReactionRow[]> {
    if (messageIds.length === 0) return [];
    return this.db('message_reactions').whereIn('message_id', messageIds);
  }

  async upsertStatus(
    trx: Knex.Transaction,
    messageId: number,
    userId: number,
    status: 'sent' | 'delivered' | 'read',
  ): Promise<void> {
    await trx('message_status')
      .insert({ message_id: messageId, user_id: userId, status })
      .onConflict(['message_id', 'user_id'])
      .merge({ status, updated_at: trx.fn.now() });
  }

  async markRead(trx: Knex.Transaction, messageIds: number[], userId: number): Promise<void> {
    if (messageIds.length === 0) return;
    const rows = messageIds.map((mid) => ({
      message_id: mid,
      user_id: userId,
      status: 'read' as const,
    }));
    await trx('message_status')
      .insert(rows)
      .onConflict(['message_id', 'user_id'])
      .merge({ status: 'read', updated_at: trx.fn.now() });
  }

    async setPinned(
    trx: Knex.Transaction,
    tenantId: number,
    messageId: number,
    pinnedById: number | null,
  ): Promise<void> {
    await trx('messages')
      .where({ tenant_id: tenantId, id: messageId })
      .update({
        pinned_at: pinnedById ? trx.fn.now() : null,
        pinned_by: pinnedById,
      });
  }

  async listPinned(
    tenantId: number,
    conversationId: number,
  ): Promise<MessageRow[]> {
    return this.db('messages')
      .where({ tenant_id: tenantId, conversation_id: conversationId })
      .whereNotNull('pinned_at')
      .orderBy('pinned_at', 'desc');
  }
}

import type { FastifyPluginAsync } from 'fastify';
import crypto from 'node:crypto';

import { env } from '../../config/env.js';
import { authenticate } from '../../middleware/authenticate.js';
import { TokenService } from '../auth/token.service.js';

const turnRoutes: FastifyPluginAsync = async (fastify) => {
  const tokens = new TokenService(fastify.db, fastify.redis);
  const auth = authenticate(tokens);

  fastify.get('/turn/credentials', { preHandler: auth }, async (req) => {
    const { userId } = req.auth!;

    // No TURN configured — return STUN only.
    if (!env.TURN_URL || !env.TURN_SECRET) {
      return {
        urls: [env.STUN_URL],
        username: '',
        credential: '',
        ttl: 0,
      };
    }

    const ttl = env.TURN_TTL;
    const username = `${Math.floor(Date.now() / 1000) + ttl}:${userId}`;
    const hmac = crypto.createHmac('sha1', env.TURN_SECRET);
    hmac.update(username);
    const credential = hmac.digest('base64');

    return {
      urls: [env.STUN_URL, env.TURN_URL],
      username,
      credential,
      ttl,
    };
  });
};

export default turnRoutes;

import type { Knex } from 'knex';
import type Redis from 'ioredis';
import { randomUUID } from 'node:crypto';

import { env } from '../../config/env.js';
import { ForbiddenError, NotFoundError, ConflictError, AppError } from '../../utils/errors.js';
import { ConversationsRepository } from '../conversations/conversations.repository.js';
import { CallsRepository, type CallRow, type CallParticipantRow } from './calls.repository.js';
import type {
  CallDTO,
  CallParticipantDTO,
  CallType,
  CallStatus,
} from '@chat/shared-types';

export class CallsService {
  private readonly repo: CallsRepository;
  private readonly convos: ConversationsRepository;

  constructor(
    private readonly db: Knex,
    private readonly redis: Redis,
  ) {
    this.repo = new CallsRepository(db);
    this.convos = new ConversationsRepository(db);
  }

  // -------------------------------------------------------------------------
  // Lifecycle
  // -------------------------------------------------------------------------

  async initiate(input: {
    tenantId: number;
    conversationId: number;
    callerId: number;
    callType: CallType;
    deviceId?: string;
  }): Promise<CallDTO> {
    // Caller must be a member of the conversation.
    const member = await this.convos.getMember(input.conversationId, input.callerId);
    if (!member) throw new ForbiddenError('Not a member of this conversation');

    // Prevent multiple simultaneous calls per user.
    const active = await this.repo.findActiveForUser(input.tenantId, input.callerId);
    if (active) throw new ConflictError('You are already in a call');

    // Determine the callee list.
    const memberRows = await this.convos.listMembers(input.conversationId);
    const calleeIds = memberRows.map((m) => m.user_id).filter((id) => id !== input.callerId);
    if (calleeIds.length === 0) throw new ConflictError('No recipients');

    // For 1:1 calls, verify callee is not busy.
    if (calleeIds.length === 1) {
      const calleeActive = await this.repo.findActiveForUser(input.tenantId, calleeIds[0]!);
      if (calleeActive) throw new ConflictError('User is busy');
    }

    const callUuid = randomUUID();

    const callId = await this.db.transaction(async (trx) => {
      const id = await this.repo.create(trx, {
        callUuid,
        tenantId: input.tenantId,
        conversationId: input.conversationId,
        initiatedBy: input.callerId,
        callType: input.callType,
        status: 'initiated',
      });

      await this.repo.addParticipant(trx, {
        callId: id,
        userId: input.callerId,
        role: 'caller',
        status: 'accepted',
        deviceId: input.deviceId ?? null,
      });
      for (const callee of calleeIds) {
        await this.repo.addParticipant(trx, {
          callId: id,
          userId: callee,
          role: 'callee',
          status: 'invited',
        });
      }

      await this.repo.logEvent(trx, {
        callId: id,
        userId: input.callerId,
        eventType: 'call_initiated',
        metadata: { callType: input.callType, callees: calleeIds },
      });

      await this.repo.updateCallStatus(trx, id, { status: 'ringing' });
      await this.repo.updateParticipant(trx, id, input.callerId, { joined_at: new Date() });

      return id;
    });

    // Store ephemeral state in Redis (used for signal routing + busy checks).
    await this.redis.set(
      `call:${callId}:state`,
      JSON.stringify({ status: 'ringing', callType: input.callType, callees: calleeIds }),
      'EX',
      env.CALL_SIGNAL_TTL_SECONDS,
    );

    // Schedule timeout in the caller's API instance (best-effort; DB is authority).
    const ttlMs = env.CALL_RING_TIMEOUT_SECONDS * 1000;
    void this.scheduleTimeout(callId, input.tenantId, ttlMs);

    return this.getById(input.tenantId, callId);
  }

  async accept(input: {
    tenantId: number;
    callId: number;
    userId: number;
    deviceId?: string;
  }): Promise<CallDTO> {
    const call = await this.repo.findById(input.tenantId, input.callId);
    if (!call) throw new NotFoundError('Call not found');
    if (['ended', 'cancelled', 'rejected', 'missed', 'failed'].includes(call.status)) {
      throw new AppError('Call is no longer active', 'CALL_NOT_ACTIVE', 409);
    }

    const participant = await this.repo.findParticipant(input.callId, input.userId);
    if (!participant) throw new ForbiddenError('You are not a participant in this call');
    if (participant.status === 'accepted' || participant.status === 'connected') {
      // Idempotent
      return this.getById(input.tenantId, input.callId);
    }

    await this.db.transaction(async (trx) => {
      await this.repo.updateParticipant(trx, input.callId, input.userId, {
        status: 'accepted',
        joined_at: new Date(),
      });
      await this.repo.updateCallStatus(trx, input.callId, { status: 'connecting' });
      await this.repo.logEvent(trx, {
        callId: input.callId,
        userId: input.userId,
        eventType: 'call_accepted',
      });
    });

    await this.redis.set(
      `call:${input.callId}:state`,
      JSON.stringify({ status: 'connecting', callType: call.call_type }),
      'EX',
      env.CALL_SIGNAL_TTL_SECONDS,
    );

    return this.getById(input.tenantId, input.callId);
  }

  async markConnected(input: {
    tenantId: number;
    callId: number;
    userId: number;
  }): Promise<CallDTO> {
    const call = await this.repo.findById(input.tenantId, input.callId);
    if (!call) throw new NotFoundError('Call not found');

    const participant = await this.repo.findParticipant(input.callId, input.userId);
    if (!participant) throw new ForbiddenError('Not a participant');

    await this.db.transaction(async (trx) => {
      const patch: Record<string, unknown> = {
        status: 'connected' as CallStatus,
      };
      if (!call.connected_at) patch.connected_at = new Date();
      await this.repo.updateCallStatus(trx, input.callId, patch);
      await this.repo.updateParticipant(trx, input.callId, input.userId, {
        status: 'connected',
      });
      await this.repo.logEvent(trx, {
        callId: input.callId,
        userId: input.userId,
        eventType: 'call_connected',
      });
    });

    return this.getById(input.tenantId, input.callId);
  }

  async reject(input: {
    tenantId: number;
    callId: number;
    userId: number;
    reason?: 'rejected' | 'busy';
  }): Promise<void> {
    const call = await this.repo.findById(input.tenantId, input.callId);
    if (!call) throw new NotFoundError('Call not found');

    const participant = await this.repo.findParticipant(input.callId, input.userId);
    if (!participant) throw new ForbiddenError('Not a participant');

    const reason = input.reason ?? 'rejected';

    await this.db.transaction(async (trx) => {
      await this.repo.updateParticipant(trx, input.callId, input.userId, {
        status: reason === 'busy' ? 'rejected' : 'rejected',
        left_at: new Date(),
      });

      // For 1:1: rejecting ends the call outright.
      const parts = await this.repo.listParticipants(input.callId);
      const activeOthers = parts.filter(
        (p) =>
          p.user_id !== input.userId &&
          ['accepted', 'connected'].includes(p.status),
      );
      if (activeOthers.length === 0) {
        await this.repo.updateCallStatus(trx, input.callId, {
          status: 'rejected',
          ended_at: new Date(),
          end_reason: reason === 'busy' ? 'busy' : 'rejected',
        });
      }

      await this.repo.logEvent(trx, {
        callId: input.callId,
        userId: input.userId,
        eventType: reason === 'busy' ? 'call_busy' : 'call_rejected',
      });
    });

    await this.redis.del(`call:${input.callId}:state`);
  }

  async cancel(input: {
    tenantId: number;
    callId: number;
    userId: number;
  }): Promise<void> {
    const call = await this.repo.findById(input.tenantId, input.callId);
    if (!call) throw new NotFoundError('Call not found');
    if (call.initiated_by !== input.userId) {
      throw new ForbiddenError('Only the caller can cancel');
    }

    await this.db.transaction(async (trx) => {
      await this.repo.updateCallStatus(trx, input.callId, {
        status: 'cancelled',
        ended_at: new Date(),
        end_reason: 'user_cancelled',
      });
      const parts = await this.repo.listParticipants(input.callId);
      for (const p of parts) {
        if (p.status === 'connected' || p.status === 'accepted') continue;
        await this.repo.updateParticipant(trx, input.callId, p.user_id, {
          status: 'cancelled',
          left_at: new Date(),
        });
      }
      await this.repo.logEvent(trx, {
        callId: input.callId,
        userId: input.userId,
        eventType: 'call_cancelled',
      });
    });

    await this.redis.del(`call:${input.callId}:state`);
  }

  async end(input: {
    tenantId: number;
    callId: number;
    userId: number;
    reason?: 'completed' | 'remote_ended' | 'network_error';
  }): Promise<void> {
    const call = await this.repo.findById(input.tenantId, input.callId);
    if (!call) throw new NotFoundError('Call not found');

    const participant = await this.repo.findParticipant(input.callId, input.userId);
    if (!participant) throw new ForbiddenError('Not a participant');

    await this.db.transaction(async (trx) => {
      const now = new Date();

      // Duration from the moment the call actually connected (or started).
      const startRef = call.connected_at ?? call.started_at;
      const durationSeconds = Math.max(
        0,
        Math.round((now.getTime() - new Date(startRef).getTime()) / 1000),
      );

      await this.repo.updateParticipant(trx, input.callId, input.userId, {
        status: 'left',
        left_at: now,
        duration_seconds: durationSeconds,
      });

      // If no one else is still connected, end the call.
      const parts = await this.repo.listParticipants(input.callId);
      const stillActive = parts.filter(
        (p) =>
          p.user_id !== input.userId &&
          ['accepted', 'connected'].includes(p.status),
      );
      if (stillActive.length === 0) {
        await this.repo.updateCallStatus(trx, input.callId, {
          status: 'ended',
          ended_at: now,
          duration_seconds: durationSeconds,
          end_reason: input.reason ?? 'completed',
        });
      }

      await this.repo.logEvent(trx, {
        callId: input.callId,
        userId: input.userId,
        eventType: 'call_ended',
        metadata: { durationSeconds, reason: input.reason ?? 'completed' },
      });
    });

    await this.redis.del(`call:${input.callId}:state`);
  }

  // -------------------------------------------------------------------------
  // Queries
  // -------------------------------------------------------------------------

  async getById(tenantId: number, callId: number): Promise<CallDTO> {
    const call = await this.repo.findById(tenantId, callId);
    if (!call) throw new NotFoundError('Call not found');
    const parts = await this.repo.listParticipants(callId);
    return this.serialize(call, parts);
  }

  async history(
    tenantId: number,
    userId: number,
    limit = 30,
    cursor?: string,
  ): Promise<{ items: CallDTO[]; nextCursor: string | null; hasMore: boolean }> {
    const cursorId = cursor ? Number(Buffer.from(cursor, 'base64url').toString('utf8')) : null;
    const rows = await this.repo.history(tenantId, userId, limit, cursorId);
    const hasMore = rows.length > limit;
    const items = hasMore ? rows.slice(0, limit) : rows;

    // Load participants for each call in a single round trip.
    const callIds = items.map((r) => r.id);
    const participantsByCall = new Map<number, CallParticipantRow[]>();
    if (callIds.length > 0) {
      const parts = await this.db('call_participants').whereIn('call_id', callIds);
      for (const p of parts) {
        const list = participantsByCall.get(p.call_id) ?? [];
        list.push(p as CallParticipantRow);
        participantsByCall.set(p.call_id, list);
      }
    }

    const dtos = items.map((r) =>
      this.serialize(r as CallRow, participantsByCall.get(r.id) ?? []),
    );

    const last = items[items.length - 1];
    return {
      items: dtos,
      hasMore,
      nextCursor:
        hasMore && last ? Buffer.from(String(last.id), 'utf8').toString('base64url') : null,
    };
  }

  // -------------------------------------------------------------------------
  // Internal helpers
  // -------------------------------------------------------------------------

  private async scheduleTimeout(
    callId: number,
    tenantId: number,
    ttlMs: number,
  ): Promise<void> {
    // Best-effort timeout: on the timeout, if the call hasn't progressed,
    // mark it missed. This runs in-process only; the DB is the source of truth.
    setTimeout(async () => {
      try {
        const call = await this.repo.findById(tenantId, callId);
        if (!call) return;
        if (call.status !== 'ringing' && call.status !== 'initiated') return;

        const parts = await this.repo.listParticipants(callId);
        const anyoneAccepted = parts.some((p) => p.status === 'accepted' || p.status === 'connected');
        if (anyoneAccepted) return;

        await this.db.transaction(async (trx) => {
          await this.repo.updateCallStatus(trx, callId, {
            status: 'missed',
            ended_at: new Date(),
            end_reason: 'timeout',
          });
          for (const p of parts) {
            if (p.status === 'invited' || p.status === 'ringing') {
              await this.repo.updateParticipant(trx, callId, p.user_id, {
                status: 'missed',
                left_at: new Date(),
              });
            }
          }
          await this.repo.logEvent(trx, {
            callId,
            userId: null,
            eventType: 'call_timeout',
          });
        });
      } catch {
        /* best-effort */
      }
    }, ttlMs).unref();
  }

  private serialize(call: CallRow, parts: CallParticipantRow[]): CallDTO {
    return {
      id: call.id,
      callUuid: call.call_uuid,
      conversationId: call.conversation_id,
      initiatedBy: call.initiated_by,
      callType: call.call_type,
      status: call.status,
      startedAt: new Date(call.started_at).toISOString(),
      connectedAt: call.connected_at ? new Date(call.connected_at).toISOString() : null,
      endedAt: call.ended_at ? new Date(call.ended_at).toISOString() : null,
      durationSeconds: call.duration_seconds,
      endReason: call.end_reason,
      participants: parts.map<CallParticipantDTO>((p) => ({
        userId: p.user_id,
        role: p.role,
        status: p.status,
        invitedAt: new Date(p.invited_at).toISOString(),
        joinedAt: p.joined_at ? new Date(p.joined_at).toISOString() : null,
        leftAt: p.left_at ? new Date(p.left_at).toISOString() : null,
        durationSeconds: p.duration_seconds,
        micMuted: Boolean(p.mic_muted),
        cameraEnabled: Boolean(p.camera_enabled),
        screenSharing: Boolean(p.screen_sharing),
      })),
    };
  }
}

  async history(
    tenantId: number,
    userId: number,
    limit: number,
    cursorId: number | null,
  ): Promise<
    Array<
      CallRow & { participants_count: number; my_status: CallParticipantStatus; my_role: 'caller' | 'callee' }
    >
  > {
    const q = this.db('calls as c')
      .join('call_participants as p_me', (j) =>
        j.on('p_me.call_id', '=', 'c.id').andOn('p_me.user_id', '=', this.db.raw('?', [userId])),

      .select(
        'c.*',
        'p_me.status as my_status',
        'p_me.role as my_role',
        this.db.raw(
          '(SELECT COUNT(*) FROM call_participants WHERE call_id = c.id) as participants_count',
        ),
      )
      .orderBy('c.id', 'desc')
      .limit(limit + 1);
    if (cursorId) q.andWhere('c.id', '<', cursorId);
    return q;
  }

import type { FastifyPluginAsync } from 'fastify';

import { authenticate } from '../../middleware/authenticate.js';
import { TokenService } from '../auth/token.service.js';
import { CallsService } from './calls.service.js';
import { getIO } from '../../sockets/io.js';
import { HistoryQuerySchema, IdParamSchema, InitiateCallSchema } from './calls.schemas.js';

const callsRoutes: FastifyPluginAsync = async (fastify) => {
  const tokens = new TokenService(fastify.db, fastify.redis);
  const service = new CallsService(fastify.db, fastify.redis);
  const auth = authenticate(tokens);

  fastify.post('/calls', { preHandler: auth }, async (req, reply) => {
    const { userId, tenantId } = req.auth!;
    const body = InitiateCallSchema.parse(req.body);
    const dto = await service.initiate({
      tenantId,
      conversationId: body.conversationId,
      callerId: userId,
      callType: body.callType,
      deviceId: body.deviceId,
    });

    try {
      const io = getIO();
      // Ring the caller's other devices.
      io.to(`t:${tenantId}:u:${userId}`).emit('call:ringing', dto);
      // Ring every callee.
      for (const p of dto.participants) {
        if (p.role === 'callee') {
          io.to(`t:${tenantId}:u:${p.userId}`).emit('call:incoming', {
            call: dto,
            from: userId,
          });
        }
      }
    } catch {
      /* io not initialized */
    }

    return reply.status(201).send(dto);
  });

  fastify.get('/calls/history', { preHandler: auth }, async (req) => {
    const { userId, tenantId } = req.auth!;
    const { limit, cursor } = HistoryQuerySchema.parse(req.query);
    return service.history(tenantId, userId, limit, cursor);
  });

  fastify.get('/calls/:id', { preHandler: auth }, async (req) => {
    const { tenantId } = req.auth!;
    const { id } = IdParamSchema.parse(req.params);
    return service.getById(tenantId, id);
  });
};

export default callsRoutes;

  async listForUser(
    tenantId: number,
    userId: number,
    limit = 20,
    cursor?: string,
  ): Promise<CursorPage<ConversationDTO>> {
    const cursorId = cursor ? Number(decodeCursor(cursor)) : null;
    const rows = await this.repo.listForUser(tenantId, userId, limit, cursorId);
    const page = buildPage(rows, limit);

    const ids = page.items.map((r) => r.id);
    if (ids.length === 0) {
      return { ...page, items: [] };
    }

    // Load members
    const members = await this.db('conversation_members')
      .whereIn('conversation_id', ids)
      .whereNull('left_at')
      .select('conversation_id', 'user_id', 'role', 'joined_at', 'last_read_at');

    // Load user profiles for those members
    const userIds = Array.from(new Set(members.map((m) => m.user_id)));
    const users = await this.repo.usersByIds(tenantId, userIds);
    const userMap = new Map(users.map((u) => [u.id, u]));

    // Load last message per conversation
    const latest = await this.repo.latestMessagesForConversations(tenantId, ids);

    // Compute unread counts — messages after each member's last_read_at
    const unreadMap = new Map<number, number>();
    await Promise.all(
      page.items.map(async (r) => {
        const lr = r.last_read_at ? new Date(r.last_read_at) : null;
        const q = this.db('messages')
          .where({ tenant_id: tenantId, conversation_id: r.id })
          .whereNot('sender_id', userId)
          .count<{ c: string }>({ c: '*' });
        if (lr) q.andWhere('created_at', '>', lr);
        const [row] = await q;
        unreadMap.set(r.id, Number(row?.c ?? 0));
      }),
    );

    const membersByConv = new Map<number, MemberDTO[]>();
    for (const m of members) {
      const list = membersByConv.get(m.conversation_id) ?? [];
      const u = userMap.get(m.user_id);
      list.push({
        userId: m.user_id,
        role: m.role,
        joinedAt: new Date(m.joined_at).toISOString(),
        lastReadAt: m.last_read_at ? new Date(m.last_read_at).toISOString() : null,
        username: u?.username ?? `user${m.user_id}`,
        displayName: u?.display_name ?? `User #${m.user_id}`,
        avatarUrl: u?.avatar_url ?? null,
      });
      membersByConv.set(m.conversation_id, list);
    }

    return {
      ...page,
      items: page.items.map((r) => {
        const lm = latest.get(r.id);
        return {
          id: r.id,
          type: r.type,
          name: r.name,
          avatarUrl: r.avatar_url,
          createdBy: r.created_by,
          lastMessageAt: r.last_message_at ? new Date(r.last_message_at).toISOString() : null,
          createdAt: new Date(r.created_at).toISOString(),
          updatedAt: new Date(r.updated_at).toISOString(),
          members: membersByConv.get(r.id) ?? [],
          myRole: r.role,
          lastMessage: lm
            ? {
                id: lm.id,
                senderId: lm.sender_id,
                type: lm.type,
                content: lm.deleted_at ? null : lm.content,
                deleted: Boolean(lm.deleted_at),
                createdAt: new Date(lm.created_at).toISOString(),
              }
            : null,
          unreadCount: unreadMap.get(r.id) ?? 0,
        };
      }),
    };
  }

  async createDirect(
    tenantId: number,
    currentUserId: number,
    peerId: number,
  ): Promise<ConversationDTO> {
    if (peerId === currentUserId) throw new ConflictError('Cannot create a direct chat with yourself');

    const peer = await this.db('users')
      .where({ tenant_id: tenantId, id: peerId, is_active: 1 })
      .first();
    if (!peer) throw new NotFoundError('Peer user not found');

    const key = directKey(currentUserId, peerId);
    const existing = await this.repo.findDirectByKey(tenantId, key);
    if (existing) return this.getById(tenantId, currentUserId, existing.id);

    const convId = await this.db.transaction(async (trx) => {
      const newId = await this.repo.create(trx, {
        tenantId,
        type: 'direct',
        createdBy: currentUserId,
        directKey: key,
      });
      await this.repo.addMember(trx, { conversationId: newId, userId: currentUserId, role: 'member' });
      await this.repo.addMember(trx, { conversationId: newId, userId: peerId, role: 'member' });
      return newId;
    });

    return this.getById(tenantId, currentUserId, convId);
  }

  async createGroup(
    tenantId: number,
    currentUserId: number,
    input: { name: string; memberIds: number[]; avatarUrl?: string | null },
  ): Promise<ConversationDTO> {
    const unique = Array.from(new Set(input.memberIds.filter((id) => id !== currentUserId)));
    if (unique.length < 1) throw new ConflictError('Group needs at least one other member');

    const found = await this.db('users')
      .where({ tenant_id: tenantId, is_active: 1 })
      .whereIn('id', unique)
      .select('id');
    if (found.length !== unique.length) throw new NotFoundError('One or more members not found');

    const id = await this.db.transaction(async (trx) => {
      const convId = await this.repo.create(trx, {
        tenantId,
        type: 'group',
        name: input.name,
        avatarUrl: input.avatarUrl ?? null,
        createdBy: currentUserId,
      });
      await this.repo.addMember(trx, { conversationId: convId, userId: currentUserId, role: 'owner' });
      for (const uid of unique) {
        await this.repo.addMember(trx, { conversationId: convId, userId: uid, role: 'member' });
      }
      return convId;
    });

    return this.getById(tenantId, currentUserId, id);

  async update(
    tenantId: number,

    if (conv.type !== 'group') throw new ConflictError('Only group conversations can be edited');

    const member = await this.repo.getMember(id, userId);
    if (!member) throw new ForbiddenError('Not a member');
    if (member.role !== 'owner' && member.role !== 'admin') {

  async addMembers(
    tenantId: number,
    userId: number,
    id: number,
    memberIds: number[],
  ): Promise<ConversationDTO> {
    const conv = await this.repo.findById(tenantId, id);
    if (!conv) throw new NotFoundError('Conversation not found');
    if (conv.type !== 'group') throw new ConflictError('Only group conversations support members');

    const me = await this.repo.getMember(id, userId);
    if (!me) throw new ForbiddenError('Not a member');
    if (me.role === 'member') throw new ForbiddenError('Insufficient permissions');

    const unique = Array.from(new Set(memberIds));
    if (unique.length === 0) return this.getById(tenantId, userId, id);

    const found = await this.db('users')
      .where({ tenant_id: tenantId, is_active: 1 })
      .whereIn('id', unique)
      .select('id');
    if (found.length !== unique.length) throw new NotFoundError('One or more members not found');

    await this.db.transaction(async (trx) => {
      for (const uid of unique) {
        await this.repo.addMember(trx, { conversationId: id, userId: uid, role: 'member' });
      }
    });

    return this.getById(tenantId, userId, id);
  }

  async removeMember(
    tenantId: number,
    userId: number,
    id: number,
    targetUserId: number,
  ): Promise<void> {
    const conv = await this.repo.findById(tenantId, id);
    if (!conv) throw new NotFoundError('Conversation not found');
    if (conv.type !== 'group') throw new ConflictError('Only group conversations support this');

    const me = await this.repo.getMember(id, userId);
    if (!me) throw new ForbiddenError('Not a member');

    const target = await this.repo.getMember(id, targetUserId);
    if (!target) throw new NotFoundError('Member not found');

    if (me.role === 'owner') {
      if (targetUserId === userId) throw new ConflictError('Use /leave to remove yourself');
    } else if (me.role === 'admin') {
      if (target.role !== 'member') throw new ForbiddenError('Admins can only remove members');
    } else {
      throw new ForbiddenError('Insufficient permissions');
    }

    await this.db.transaction(async (trx) => {
      await this.repo.removeMember(trx, id, targetUserId);
    });
  }

  async leave(tenantId: number, userId: number, id: number): Promise<void> {
    const conv = await this.repo.findById(tenantId, id);
    if (!conv) throw new NotFoundError('Conversation not found');
    if (conv.type !== 'group') throw new ConflictError('Direct conversations cannot be left');

    const me = await this.repo.getMember(id, userId);
    if (!me) throw new ForbiddenError('Not a member');

    await this.db.transaction(async (trx) => {
      await this.repo.removeMember(trx, id, userId);

      if (me.role === 'owner') {
        const remaining = await trx('conversation_members')
          .where({ conversation_id: id })
          .whereNull('left_at')
          .orderBy('joined_at', 'asc')
          .first();
        if (remaining) {
          await trx('conversation_members')
            .where({ id: remaining.id })
            .update({ role: 'owner' });
        }
      }
    });
  }

  async promote(
    tenantId: number,
    userId: number,
    id: number,
    targetUserId: number,
    role: 'admin' | 'member',
  ): Promise<void> {
    const conv = await this.repo.findById(tenantId, id);
    if (!conv || conv.type !== 'group') throw new NotFoundError('Group not found');

    const me = await this.repo.getMember(id, userId);
    if (!me) throw new ForbiddenError('Not a member');
    if (me.role !== 'owner') throw new ForbiddenError('Only owner can change roles');

    const target = await this.repo.getMember(id, targetUserId);
    if (!target) throw new NotFoundError('Member not found');
    if (target.user_id === userId) {
      throw new ConflictError('Cannot change own role');
    }
    await this.db.transaction(async (trx) => {
      await this.repo.updateMemberRole(trx, id, targetUserId, role);
    });
  }
}

      .where({ conversation_id: conversationId, user_id: userId })
      .update({ role });
  }

    tenantId: number,
    userId: number,
    limit: number,
    cursorId: number | null,
  ): Promise<Array<ConversationRow & { last_read_at: Date | null; role: string }>> {
    const q = this.db('conversations as c')
      .join('conversation_members as m', 'm.conversation_id', 'c.id')
      .where('c.tenant_id', tenantId)
      .where('m.user_id', userId)
      .whereNull('m.left_at')
      .select('c.*', 'm.last_read_at as last_read_at', 'm.role as role')
      .orderBy('c.last_message_at', 'desc')
      .orderBy('c.id', 'desc')
      .limit(limit + 1);

    if (cursorId) q.andWhere('c.id', '<', cursorId);
    return q;
  }


  async latestMessagesForConversations(
    tenantId: number,
    conversationIds: number[],
  ): Promise<Map<number, LatestMessage>> {
    const map = new Map<number, LatestMessage>();

    if (conversationIds.length === 0) return map;

    // Build a subquery to rank messages within each conversation.
    const ranked = this.db('messages')
      .select(
        'id',
        'conversation_id',
        'sender_id',
        'type',
        'content',
        'deleted_at',
        'created_at',
      )
      .select(
        this.db.raw(
          'ROW_NUMBER() OVER (PARTITION BY conversation_id ORDER BY id DESC) AS rn',
        ),
      )
      .whereIn('conversation_id', conversationIds)
      .where('tenant_id', tenantId)
      .as('ranked');

    // Get the latest message from each conversation.
    const rows = await this.db
      .from<LatestMessage & { rn: number }>(ranked)
      .select('*')
      .where('rn', 1);

    for (const r of rows) {
      map.set(r.conversation_id, {
        conversation_id: r.conversation_id,
        id: r.id,
        sender_id: r.sender_id,
        type: r.type,
        content: r.content,
        deleted_at: r.deleted_at,
        created_at: r.created_at,
      });
    }

    return map;
  }


  /** Resolve multiple users (for names) in one query. */
  async usersByIds(
    tenantId: number,
    userIds: number[],
  ): Promise<Array<{ id: number; username: string; display_name: string; avatar_url: string | null }>> {
    if (userIds.length === 0) return [];
    return this.db('users')
      .where({ tenant_id: tenantId })
      .whereIn('id', userIds)
      .select('id', 'username', 'display_name', 'avatar_url');
  }

  async updateConversation(
    tenantId: number,
    id: number,
    patch: { name?: string | null; avatar_url?: string | null },
  ): Promise<void> {
    const clean: Record<string, unknown> = {};
    if (patch.name !== undefined) clean.name = patch.name;
    if (patch.avatar_url !== undefined) clean.avatar_url = patch.avatar_url;
    if (Object.keys(clean).length === 0) return;
    clean.updated_at = this.db.fn.now();
    await this.db('conversations').where({ tenant_id: tenantId, id }).update(clean);
  }

  async touch(
    trx: Knex.Transaction,
    tenantId: number,
    conversationId: number,
  ): Promise<void> {
    await trx('conversations')
      .where({ tenant_id: tenantId, id: conversationId })
      .update({ last_message_at: trx.fn.now(), updated_at: trx.fn.now() });
  }
}

import type { Knex } from 'knex';
import type Redis from 'ioredis';
import { randomUUID } from 'node:crypto';

import { AppError, ConflictError, UnauthorizedError } from '../../utils/errors.js';
import { PasswordService } from './password.service.js';
import { TokenService } from './token.service.js';

export interface AuthContext {
  userAgent?: string;
  ip?: string;
  deviceId?: string;
}

export interface IssuedTokens {
  accessToken: string;
  refreshToken: string;
  accessExpiresIn: number;
  user: PublicUser;
}

export interface PublicUser {
  id: number;
  tenantId: number;
  username: string;
  displayName: string;
  email: string | null;
  avatarUrl: string | null;
  role: string;
}

interface UserRow {
  id: number;
  tenant_id: number;
  external_id: string | null;
  username: string;
  display_name: string;
  email: string | null;
  password_hash: string | null;
  avatar_url: string | null;
  role: string;
  is_active: 0 | 1;
}

export class AuthService {
  private readonly tokens: TokenService;

  constructor(
    private readonly db: Knex,
    private readonly redis: Redis,
  ) {
    this.tokens = new TokenService(db, redis);
  }

  get tokenService(): TokenService {
    return this.tokens;
  }

  // ---------- Register (email/password) ----------
  async register(input: {
    tenantId: number;
    username: string;
    email?: string;
    password: string;
    displayName?: string;
  }, ctx: AuthContext): Promise<IssuedTokens> {
    const username = input.username.trim().toLowerCase();
    const email = input.email?.trim().toLowerCase() || null;

    const existing = await this.db('users')
      .where({ tenant_id: input.tenantId, username })
      .first();
    if (existing) throw new ConflictError('Username already taken');

    if (email) {
      const existingEmail = await this.db('users')
        .where({ tenant_id: input.tenantId, email })
        .first();
      if (existingEmail) throw new ConflictError('Email already registered');
    }

    const passwordHash = await PasswordService.hash(input.password);

    const trxResult = await this.db.transaction(async (trx) => {
      const [id] = await trx('users').insert({
        tenant_id: input.tenantId,
        username,
        email,
        display_name: input.displayName?.trim() || username,
        password_hash: passwordHash,
        role: 'user',
      });
      await trx('user_settings').insert({ user_id: id });
      return Number(id);
    });

    const user = await this.getUserOrThrow(trxResult);
    return this.issueSession(user, ctx);
  }

  // ---------- Login (email/password) ----------
  async login(input: {
    tenantId: number;
    identifier: string; // username or email
    password: string;
  }, ctx: AuthContext): Promise<IssuedTokens> {
    const identifier = input.identifier.trim().toLowerCase();

    const row: UserRow | undefined = await this.db('users')
      .where({ tenant_id: input.tenantId })
      .andWhere((b) => b.where('username', identifier).orWhere('email', identifier))
      .first();

    if (!row || !row.password_hash) {
      throw new UnauthorizedError('Invalid credentials');
    }
    if (!row.is_active) {
      throw new UnauthorizedError('Account disabled');
    }
    const ok = await PasswordService.verify(row.password_hash, input.password);
    if (!ok) throw new UnauthorizedError('Invalid credentials');

    const user = await this.getUserOrThrow(row.id);
    return this.issueSession(user, ctx);
  }

  // ---------- Widget login ----------
  async widgetLogin(widgetToken: string, ctx: AuthContext): Promise<IssuedTokens> {
    const claims = await this.tokens.verifyWidgetToken(widgetToken);

    const tenant = await this.db('tenants').where({ slug: claims.tenantSlug }).first();
    if (!tenant || !tenant.is_active) {
      throw new UnauthorizedError('Tenant not found or inactive');
    }

    const externalId = String(claims.sub);
    let row: UserRow | undefined = await this.db('users')
      .where({ tenant_id: tenant.id, external_id: externalId })
      .first();

    if (!row) {
      const username = `u_${externalId}`.slice(0, 64).toLowerCase();
      const trxId = await this.db.transaction(async (trx) => {
        const [id] = await trx('users').insert({
          tenant_id: tenant.id,
          external_id: externalId,
          username,
          display_name: claims.name ?? username,
          email: claims.email ?? null,
          role: 'user',
        });
        await trx('user_settings').insert({ user_id: id });
        return Number(id);
      });
      row = (await this.db('users').where({ id: trxId }).first()) as UserRow;
    } else {
      // Refresh profile fields on each login
      await this.db('users').where({ id: row.id }).update({
        display_name: claims.name ?? row.display_name,
        email: claims.email ?? row.email,
        updated_at: this.db.fn.now(),
      });
    }

    const user = await this.getUserOrThrow(row.id);
    return this.issueSession(user, ctx);
  }

  // ---------- Refresh ----------
  async refresh(refreshToken: string, ctx: AuthContext): Promise<IssuedTokens> {
    const record = await this.tokens.verifyRefreshToken(refreshToken);
    const user = await this.getUserOrThrow(record.userId);

    // Rotation
    const next = await this.tokens.rotateRefreshToken(record, {
      deviceId: ctx.deviceId,
      userAgent: ctx.userAgent,
      ip: ctx.ip,
    });

    const access = await this.tokens.issueAccessToken({
      userId: user.id,
      tenantId: user.tenantId,
      role: user.role,
    });

    return {
      accessToken: access.token,
      refreshToken: next.token,
      accessExpiresIn: access.expiresIn,
      user,
    };
  }

  // ---------- Logout ----------
  async logout(input: { refreshToken?: string; accessJti?: string; accessTtlSeconds?: number }): Promise<void> {
    if (input.refreshToken) {
      await this.tokens.revokeRefreshToken(input.refreshToken);
    }
    if (input.accessJti && input.accessTtlSeconds && input.accessTtlSeconds > 0) {
      await this.tokens.revokeJti(input.accessJti, input.accessTtlSeconds);
    }
  }

  // ---------- Helpers ----------
  private async issueSession(user: PublicUser, ctx: AuthContext): Promise<IssuedTokens> {
    const access = await this.tokens.issueAccessToken({
      userId: user.id,
      tenantId: user.tenantId,
      role: user.role,
    });
    const refresh = await this.tokens.issueRefreshToken({
      userId: user.id,
      deviceId: ctx.deviceId,
      userAgent: ctx.userAgent,
      ip: ctx.ip,
    });

    // Track device if deviceId provided
    if (ctx.deviceId) {
      await this.db('user_devices')
        .insert({
          user_id: user.id,
          device_id: ctx.deviceId,
          platform: 'web',
          user_agent: ctx.userAgent ?? null,
          last_active_at: this.db.fn.now(),
        })
        .onConflict(['user_id', 'device_id'])
        .merge({ last_active_at: this.db.fn.now(), user_agent: ctx.userAgent ?? null });
    }

    return {
      accessToken: access.token,
      refreshToken: refresh.token,
      accessExpiresIn: access.expiresIn,
      user,
    };
  }

  private async getUserOrThrow(id: number): Promise<PublicUser> {
    const row: UserRow | undefined = await this.db('users').where({ id }).first();
    if (!row) throw new AppError('User not found', 'USER_NOT_FOUND', 404);
    if (!row.is_active) throw new UnauthorizedError('Account disabled');
    return {
      id: row.id,
      tenantId: row.tenant_id,
      username: row.username,
      displayName: row.display_name,
      email: row.email,
      avatarUrl: row.avatar_url,
      role: row.role,
    };
  }
}

import { SignJWT, jwtVerify, type JWTPayload } from 'jose';
import crypto from 'node:crypto';
import type { Knex } from 'knex';
import type Redis from 'ioredis';

import { env } from '../../config/env.js';
import { UnauthorizedError } from '../../utils/errors.js';

const ACCESS_SECRET = new TextEncoder().encode(env.JWT_ACCESS_SECRET);
const WIDGET_SECRET = new TextEncoder().encode(env.WIDGET_SHARED_SECRET);

export interface AccessTokenClaims extends JWTPayload {
  sub: string;      // userId
  tenantId: number;
  role: string;
  type: 'access';
}

export interface WidgetTokenClaims extends JWTPayload {
  sub: string;      // external user id
  appId: string;
  tenantSlug: string;
  name?: string;
  email?: string;
}

export interface RefreshRecord {
  id: number;
  userId: number;
  tokenHash: string;
  expiresAt: Date;
}

export class TokenService {
  constructor(
    private readonly db: Knex,
    private readonly redis: Redis,
  ) {}

  // ---------- Access tokens (JWT) ----------
  async issueAccessToken(input: {
    userId: number;
    tenantId: number;
    role: string;
  }): Promise<{ token: string; expiresIn: number; jti: string }> {
    const jti = crypto.randomUUID();
    const token = await new SignJWT({
      tenantId: input.tenantId,
      role: input.role,
      type: 'access',
    })
      .setProtectedHeader({ alg: 'HS256', typ: 'JWT' })
      .setSubject(String(input.userId))
      .setJti(jti)
      .setIssuedAt()
      .setExpirationTime(`${env.JWT_ACCESS_TTL}s`)
      .sign(ACCESS_SECRET);

    return { token, expiresIn: env.JWT_ACCESS_TTL, jti };
  }

  async verifyAccessToken(token: string): Promise<AccessTokenClaims> {
    try {
      const { payload } = await jwtVerify(token, ACCESS_SECRET);
      if (payload.type !== 'access' || !payload.sub || !payload.jti) {
        throw new UnauthorizedError('Invalid token payload');
      }
      // Revocation check
      const revoked = await this.redis.get(`revoked:jti:${payload.jti}`);
      if (revoked) throw new UnauthorizedError('Token revoked');
      return payload as AccessTokenClaims;
    } catch (err) {
      if (err instanceof UnauthorizedError) throw err;
      throw new UnauthorizedError('Invalid or expired token');
    }
  }

  async revokeJti(jti: string, ttlSeconds: number): Promise<void> {
    if (ttlSeconds > 0) {
      await this.redis.set(`revoked:jti:${jti}`, '1', 'EX', ttlSeconds);
    }
  }

  // ---------- Refresh tokens (opaque, hashed in DB) ----------
  async issueRefreshToken(input: {
    userId: number;
    deviceId?: string;
    userAgent?: string;
    ip?: string;
  }): Promise<{ token: string; record: RefreshRecord }> {
    const raw = crypto.randomBytes(48).toString('base64url');
    const hash = this.hashRefresh(raw);
    const expiresAt = new Date(Date.now() + env.JWT_REFRESH_TTL * 1000);

    const [id] = await this.db('refresh_tokens').insert({
      user_id: input.userId,
      token_hash: hash,
      device_id: input.deviceId ?? null,
      user_agent: input.userAgent ?? null,
      ip_address: input.ip ?? null,
      expires_at: expiresAt,
    });

    return {
      token: raw,
      record: { id: Number(id), userId: input.userId, tokenHash: hash, expiresAt },
    };
  }

  async verifyRefreshToken(raw: string): Promise<RefreshRecord> {
    const hash = this.hashRefresh(raw);
    const row = await this.db('refresh_tokens')
      .where({ token_hash: hash })
      .whereNull('revoked_at')
      .first();

    if (!row) throw new UnauthorizedError('Invalid refresh token');
    if (new Date(row.expires_at).getTime() < Date.now()) {
      throw new UnauthorizedError('Refresh token expired');
    }
    return {
      id: row.id,
      userId: row.user_id,
      tokenHash: row.token_hash,
      expiresAt: new Date(row.expires_at),
    };
  }

  async rotateRefreshToken(
    oldRecord: RefreshRecord,
    input: { deviceId?: string; userAgent?: string; ip?: string },
  ): Promise<{ token: string; record: RefreshRecord }> {
    const next = await this.issueRefreshToken({
      userId: oldRecord.userId,
      ...input,
    });
    await this.db('refresh_tokens')
      .where({ id: oldRecord.id })
      .update({ revoked_at: this.db.fn.now(), replaced_by_id: next.record.id });
    return next;
  }

  async revokeRefreshToken(raw: string): Promise<void> {
    const hash = this.hashRefresh(raw);
    await this.db('refresh_tokens')
      .where({ token_hash: hash })
      .whereNull('revoked_at')
      .update({ revoked_at: this.db.fn.now() });
  }

  async revokeAllForUser(userId: number): Promise<void> {
    await this.db('refresh_tokens')
      .where({ user_id: userId })
      .whereNull('revoked_at')
      .update({ revoked_at: this.db.fn.now() });
  }

  // ---------- Widget tokens ----------
  async verifyWidgetToken(token: string): Promise<WidgetTokenClaims> {
    try {
      const { payload } = await jwtVerify(token, WIDGET_SECRET, {
        algorithms: ['HS256'],
      });
      if (!payload.sub || !payload.appId || !payload.tenantSlug) {
        throw new UnauthorizedError('Invalid widget token claims');
      }
      return payload as WidgetTokenClaims;
    } catch (err) {
      if (err instanceof UnauthorizedError) throw err;
      throw new UnauthorizedError('Invalid widget token');
    }
  }

  private hashRefresh(raw: string): string {
    return crypto.createHash('sha256').update(raw).digest('hex');
  }
}

import type { Knex } from 'knex';
import type { Readable } from 'node:stream';
import { promises as fsp } from 'node:fs';

import { ForbiddenError, NotFoundError, ValidationError } from '../../utils/errors.js';
import { allowedMimeTypes, env } from '../../config/env.js';
import { storage } from './storage.service.js';
import { AttachmentsRepository, type AttachmentRow } from './attachments.repository.js';

const MAGIC: Array<{ mime: string; bytes: number[]; offset?: number }> = [
  { mime: 'image/jpeg', bytes: [0xff, 0xd8, 0xff] },
  { mime: 'image/png', bytes: [0x89, 0x50, 0x4e, 0x47] },
  { mime: 'image/gif', bytes: [0x47, 0x49, 0x46, 0x38] },
  { mime: 'image/webp', bytes: [0x52, 0x49, 0x46, 0x46], offset: 0 },
  { mime: 'application/pdf', bytes: [0x25, 0x50, 0x44, 0x46] },
  { mime: 'video/mp4', bytes: [0x66, 0x74, 0x79, 0x70], offset: 4 },
  { mime: 'audio/mpeg', bytes: [0x49, 0x44, 0x33] },
  { mime: 'audio/ogg', bytes: [0x4f, 0x67, 0x67, 0x53] },
];

function sniffMime(header: Buffer): string | null {
  for (const sig of MAGIC) {
    const off = sig.offset ?? 0;
    if (header.length < off + sig.bytes.length) continue;
    let ok = true;
    for (let i = 0; i < sig.bytes.length; i++) {
      if (header[off + i] !== sig.bytes[i]) { ok = false; break; }
    }
    if (ok) return sig.mime;
  }
  return null;
}

export interface AttachmentDTO {
  id: number;
  messageId: number | null;
  mimeType: string;
  sizeBytes: number;
  originalName: string | null;
  url: string;
  metadata: Record<string, unknown> | null;
}

export class AttachmentsService {
  private readonly repo: AttachmentsRepository;

  constructor(private readonly db: Knex) {
    this.repo = new AttachmentsRepository(db);
  }

  async upload(input: {
    tenantId: number;
    userId: number;
    stream: Readable;
    originalName: string;
    declaredMime: string;
  }): Promise<AttachmentDTO> {
    const declared = (input.declaredMime || 'application/octet-stream').toLowerCase();
    if (!allowedMimeTypes.includes(declared)) {
      throw new ValidationError(`MIME type not allowed: ${declared}`);
    }

    const stored = await storage.save({
      tenantId: input.tenantId,
      stream: input.stream,
      originalName: input.originalName,
      mimeType: declared,
    });

    if (stored.size > env.UPLOAD_MAX_BYTES) {
      await storage.delete(stored.key);
      throw new ValidationError('File too large');
    }

    const fh = await fsp.open(stored.absPath, 'r');
    const buf = Buffer.alloc(512);
    try { await fh.read(buf, 0, 512, 0); } finally { await fh.close(); }
    const sniffed = sniffMime(buf);
    if (sniffed && !allowedMimeTypes.includes(sniffed)) {
      await storage.delete(stored.key);
      throw new ValidationError(`File content does not match allowed types (${sniffed})`);
    }

    const id = await this.db.transaction(async (trx) => {
      return this.repo.insert(trx, {
        messageId: null,
        tenantId: input.tenantId,
        uploadedBy: input.userId,
        storageKey: stored.key,
        mimeType: sniffed ?? declared,
        sizeBytes: stored.size,
        originalName: input.originalName.slice(0, 255),
        metadata: { sha256: stored.sha256 },
      });
    });

    return {
      id,
      messageId: null,
      mimeType: sniffed ?? declared,
      sizeBytes: stored.size,
      originalName: input.originalName,
      url: `${env.PUBLIC_API_BASE_URL}/api/v1/attachments/${id}`,
      metadata: { sha256: stored.sha256 },
    };
  }

  async attachToMessage(
    trx: Knex.Transaction,
    tenantId: number,
    messageId: number,
    attachmentIds: number[],
  ): Promise<void> {
    if (attachmentIds.length === 0) return;

    const rows: AttachmentRow[] = await trx('attachments')
      .where({ tenant_id: tenantId })
      .whereIn('id', attachmentIds);

    if (rows.length !== attachmentIds.length) throw new NotFoundError('One or more attachments not found');
    for (const r of rows) {
      if (r.message_id !== null) throw new ForbiddenError('Attachment already attached');
    }
    await this.repo.attachToMessage(trx, attachmentIds, messageId, tenantId);
  }

  async loadForMessages(tenantId: number, messageIds: number[]): Promise<Map<number, AttachmentDTO[]>> {
    const map = new Map<number, AttachmentDTO[]>();
    if (messageIds.length === 0) return map;
    const rows: AttachmentRow[] = await this.db('attachments')
      .where({ tenant_id: tenantId })
      .whereIn('message_id', messageIds);
    for (const r of rows) {
      if (r.message_id == null) continue;
      const arr = map.get(r.message_id) ?? [];
      arr.push(this.toDTO(r));
      map.set(r.message_id, arr);
    }
    return map;
  }

  async getForDownload(tenantId: number, id: number): Promise<{
    absPath: string; mimeType: string; sizeBytes: number; originalName: string | null;
  }> {
    const row = await this.repo.findById(tenantId, id);
    if (!row) throw new NotFoundError('Attachment not found');
    return {
      absPath: storage.absPath(row.storage_key),
      mimeType: row.mime_type,
      sizeBytes: Number(row.size_bytes),
      originalName: row.original_name,
    };
  }

  toDTO(r: AttachmentRow): AttachmentDTO {
    return {
      id: r.id,
      messageId: r.message_id,
      mimeType: r.mime_type,
      sizeBytes: Number(r.size_bytes),
      originalName: r.original_name,
      url: `${env.PUBLIC_API_BASE_URL}/api/v1/attachments/${r.id}`,
      metadata: (r.metadata as Record<string, unknown>) ?? null,
    };
  }
}

import type { FastifyReply, FastifyRequest } from 'fastify';
import { UnauthorizedError } from '../utils/errors.js';
import type { TokenService } from '../modules/auth/token.service.js';

declare module 'fastify' {
  interface FastifyRequest {
    auth?: {
      userId: number;
      tenantId: number;
      role: string;
      jti: string;
      raw: string;
    };
  }
}

export function authenticate(tokens: TokenService) {
  return async function authenticateHook(
    request: FastifyRequest,
    _reply: FastifyReply,
  ): Promise<void> {
    const header = request.headers.authorization;
    if (!header?.startsWith('Bearer ')) {
      throw new UnauthorizedError('Missing bearer token');
    }
    const raw = header.slice('Bearer '.length).trim();
    const claims = await tokens.verifyAccessToken(raw);
    request.auth = {
      userId: Number(claims.sub),
      tenantId: Number(claims.tenantId),
      role: String(claims.role),
      jti: String(claims.jti),
      raw,
    };
  };
}

import Fastify, { type FastifyInstance } from 'fastify';
import helmet from '@fastify/helmet';
import rateLimit from '@fastify/rate-limit';
import sensible from '@fastify/sensible';
import underPressure from '@fastify/under-pressure';
import multipart from '@fastify/multipart';

import { env, isProduction } from './config/env.js';
import { logger } from './config/logger.js';
import dbPlugin from './plugins/db.plugin.js';
import redisPlugin from './plugins/redis.plugin.js';
import corsPlugin from './plugins/cors.plugin.js';
import errorHandlerPlugin from './plugins/error-handler.plugin.js';
import routes from './routes/index.js';
import { initSocketIO } from './sockets/io.js';

export async function buildApp(): Promise<FastifyInstance> {
  const app = Fastify({
    logger: true,
    trustProxy: true,
    bodyLimit: 1 * 1024 * 1024,
    requestIdHeader: 'x-request-id',
    genReqId: () => crypto.randomUUID(),
  });

  await app.register(helmet, {
    contentSecurityPolicy: isProduction ? undefined : false,
    crossOriginEmbedderPolicy: false,
    crossOriginResourcePolicy: { policy: 'cross-origin' },
  });

  await app.register(sensible);

  await app.register(multipart, {
    limits: {
      fileSize: env.UPLOAD_MAX_BYTES,
      files: 5,
      fields: 20,
    },
    attachFieldsToBody: false,
  });

  await app.register(rateLimit, {
    max: env.RATE_LIMIT_MAX,
    timeWindow: env.RATE_LIMIT_WINDOW_MS,
    allowList: [],
    keyGenerator: (req) => req.ip,
  });

  await app.register(underPressure, {
    maxEventLoopDelay: 1000,
    maxHeapUsedBytes: 0,
    maxRssBytes: 0,
    maxEventLoopUtilization: 0.98,
    message: 'Service under pressure',
  });

  await app.register(dbPlugin);
  await app.register(redisPlugin);
  await app.register(corsPlugin);
  await app.register(errorHandlerPlugin);

  await app.register(routes, { prefix: '/api/v1' });

    // Accept empty bodies with application/json content-type
  app.addContentTypeParser(
    'application/json',
    { parseAs: 'string' },
    (_req, body, done) => {
      if (!body || (typeof body === 'string' && body.trim().length === 0)) {
        done(null, {});
        return;
      }
      try {
        done(null, JSON.parse(body as string));
      } catch (err) {
        done(err as Error, undefined);
      }
    },
  );

  await initSocketIO(app);

  return app;
}
